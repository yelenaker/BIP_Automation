  *   Gmail -> Gemini AI -> Parser -> Google Sheets -> Google Drive
  *   -> Reminder Emails -> Settings Web App.
  *
  * ARCHITECTURAL POSITION (Layer 0 - Foundation)
  * This file has NO dependencies on any other file in the project.
  * Every other file (Utilities, Config, and every Service) depends on it,
  * directly or indirectly. It must never contain logic, I/O, or side
  * effects - only literal values.
  *
  * GLOBAL SCOPE NOTE
  * Google Apps Script executes every .gs file in one shared global
  * namespace (there is no module system). To avoid namespace collisions
  * across 15 files, this file exposes exactly ONE global symbol,
  * `BIP_CONSTANTS`, built via an IIFE. All nested groups are frozen
  * individually before being composed, and the composed result is frozen
  * again, giving effective deep immutability. Consumers reference values
  * as `BIP_CONSTANTS.SHEETS.MAIN`, `BIP_CONSTANTS.ROUTER_ACTIONS.IMPORT`,
  * etc. Never declare a second top-level `const`/`var` with a generic
  * name (e.g. `SHEETS`, `CONFIG`) in any other file - it would collide
  * with every other file sharing this global scope.
  *

  * `BIP_FIELD_KEYS` defines internal, stable field identifiers used
  * throughout the codebase (Parser DTOs, Sheet writes, email merge tags).

  * column headers and the *actual* `Email Templates` template names,
  * which are not yet known. Those literal mappings will be added to this
  * file when `SpreadsheetService.gs` and `NotificationService.gs` are
  * built, once the real sheet contents are confirmed. No code elsewhere
  * will need to change when that happens - only the mapping data here.

Utilities.gs

  /**
  * @fileoverview Shared utilities: logging, validation, response building, and error handling.
  */

  const BIP_UTILS = Object.freeze((() => {

    class AppError extends Error {
      /**
      * @param {string} message
      * @param {string} errorCode
      * @param {Object=} details
      */
      constructor(message, errorCode, details) {
        super(message);
        this.name = 'AppError';
        this.errorCode = errorCode || BIP_CONSTANTS.ERROR_CODES.INTERNAL_ERROR;
        this.details = details || null;
      }
    }

    /**
    * @param {string} level
    * @param {string} message
    * @param {Object=} context
    */
    function logMessage(level, message, context) {
      const timestamp = new Date().toISOString();
      const suffix = context ? ` | ${JSON.stringify(context)}` : '';
      const line = `[${timestamp}] [${level}] ${message}${suffix}`;
      switch (level) {
        case BIP_CONSTANTS.LOG_LEVELS.ERROR:
          console.error(line);
          break;
        case BIP_CONSTANTS.LOG_LEVELS.WARN:
          console.warn(line);
          break;
        default:
          console.log(line);
      }
    }

    /**
    * @param {string} message
    * @param {Object=} context
    */
    function logDebug(message, context) {
      logMessage(BIP_CONSTANTS.LOG_LEVELS.DEBUG, message, context);
    }

    /**
    * @param {string} message
    * @param {Object=} context
    */
    function logInfo(message, context) {
      logMessage(BIP_CONSTANTS.LOG_LEVELS.INFO, message, context);
    }

    /**
    * @param {string} message
    * @param {Object=} context
    */
    function logWarn(message, context) {
      logMessage(BIP_CONSTANTS.LOG_LEVELS.WARN, message, context);
    }

    /**
    * @param {string} message
    * @param {Object=} context
    */
    function logError(message, context) {
      logMessage(BIP_CONSTANTS.LOG_LEVELS.ERROR, message, context);
    }

    /**
    * @param {*} value
    * @return {boolean}
    */
    function isNonEmptyString(value) {
      return typeof value === 'string' && value.trim().length > 0;
    }

    /**
    * @param {*} value
    * @return {boolean}
    */
    function isPlainObject(value) {
      return typeof value === 'object' && value !== null && !Array.isArray(value);
    }

    /**
    * @param {*} value
    * @param {string} fieldName
    * @return {string}
    * @throws {AppError}
    */
    function requireNonEmptyString(value, fieldName) {
      if (!isNonEmptyString(value)) {
        throw new AppError(
          `Field "${fieldName}" is required and cannot be empty.`,
          BIP_CONSTANTS.ERROR_CODES.VALIDATION_ERROR,
          { field: fieldName }
        );
      }
      return value.trim();
    }

    /**
    * @param {*} value
    * @param {string} fieldName
    * @return {*}
    * @throws {AppError}
    */
    function requireNonNull(value, fieldName) {
      if (value === null || value === undefined) {
        throw new AppError(
          `Field "${fieldName}" is required.`,
          BIP_CONSTANTS.ERROR_CODES.VALIDATION_ERROR,
          { field: fieldName }
        );
      }
      return value;
    }

    /**
    * @param {string} email
    * @return {boolean}
    */
    function isValidEmail(email) {
      return isNonEmptyString(email) && BIP_CONSTANTS.VALIDATION_PATTERNS.EMAIL.test(email.trim());
    }

    /**
    * @param {string} email
    * @param {string} fieldName
    * @return {string}
    * @throws {AppError}
    */
    function requireValidEmail(email, fieldName) {
      if (!isValidEmail(email)) {
        throw new AppError(
          `Field "${fieldName}" must be a valid email address.`,
          BIP_CONSTANTS.ERROR_CODES.VALIDATION_ERROR,
          { field: fieldName, value: email }
        );
      }
      return email.trim();
    }

    /**
    * @param {string} dateString
    * @return {boolean}
    */
    function isValidIsoDate(dateString) {
      return isNonEmptyString(dateString)
        && BIP_CONSTANTS.VALIDATION_PATTERNS.ISO_DATE.test(dateString.trim())
        && !isNaN(new Date(dateString).getTime());
    }

    /**
    * @param {string} dateString
    * @param {string} fieldName
    * @return {string}
    * @throws {AppError}
    */
    function requireValidIsoDate(dateString, fieldName) {
      if (!isValidIsoDate(dateString)) {
        throw new AppError(
          `Field "${fieldName}" must be a valid ISO date (yyyy-MM-dd).`,
          BIP_CONSTANTS.ERROR_CODES.VALIDATION_ERROR,
          { field: fieldName, value: dateString }
        );
      }
      return dateString.trim();
    }

    /**
    * @param {Date} date
    * @param {string} timeZone
    * @return {string}
    * @throws {AppError}
    */
    function formatDateIso(date, timeZone) {
      if (!(date instanceof Date) || isNaN(date.getTime())) {
        throw new AppError('formatDateIso requires a valid Date instance.', BIP_CONSTANTS.ERROR_CODES.VALIDATION_ERROR);
      }
      return Utilities.formatDate(date, timeZone, 'yyyy-MM-dd');
    }

    /**
    * @param {string} dateString
    * @return {?Date}
    */
    function parseIsoDate(dateString) {
      if (!isValidIsoDate(dateString)) {
        return null;
      }
      const parsed = new Date(`${dateString.trim()}T00:00:00`);
      return isNaN(parsed.getTime()) ? null : parsed;
    }

    /**
    * @return {string}
    */
    function getCurrentIsoTimestamp() {
      return new Date().toISOString();
    }

    /**
    * @param {Date} startDate
    * @param {Date} endDate
    * @return {number}
    * @throws {AppError}
    */
    function daysBetween(startDate, endDate) {
      requireNonNull(startDate, 'startDate');
      requireNonNull(endDate, 'endDate');
      return Math.floor((endDate.getTime() - startDate.getTime()) / BIP_CONSTANTS.TIME.MILLISECONDS_PER_DAY);
    }

    /**
    * @param {Date} date
    * @param {number} days
    * @return {Date}
    * @throws {AppError}
    */
    function addDays(date, days) {
      requireNonNull(date, 'date');
      const result = new Date(date.getTime());
      result.setDate(result.getDate() + days);
      return result;
    }

    /**
    * @param {string} text
    * @return {string}
    */
    function sanitizeForFileName(text) {
      const fallback = 'Untitled';
      if (!isNonEmptyString(text)) {
        return fallback;
      }
      const sanitized = text.replace(/[\\/:*?"<>|]/g, ' ').replace(/\s+/g, ' ').trim();
      return sanitized.length > 0 ? sanitized : fallback;
    }

    /**
    * @param {string} text
    * @param {number} maxLength
    * @return {string}
    */
    function truncateString(text, maxLength) {
      if (!isNonEmptyString(text)) {
        return '';
      }
      return text.length > maxLength ? `${text.substring(0, maxLength).trim()}...` : text;
    }

    /**
    * @param {string} text
    * @return {string}
    */
    function escapeHtml(text) {
      if (typeof text !== 'string') {
        return '';
      }
      return text
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
    }

    /**
    * @param {string} text
    * @param {*} fallbackValue
    * @return {*}
    */
    function safeJsonParse(text, fallbackValue) {
      try {
        return JSON.parse(text);
      } catch (parseError) {
        logWarn('Failed to parse JSON, returning fallback value.', { error: parseError.message });
        return fallbackValue;
      }
    }

    /**
    * @param {string} action
    * @param {*} data
    * @return {Object}
    */
    function buildSuccessResponse(action, data) {
      return {
        [BIP_CONSTANTS.RESPONSE_FIELDS.SUCCESS]: true,
        [BIP_CONSTANTS.RESPONSE_FIELDS.ACTION]: action,
        [BIP_CONSTANTS.RESPONSE_FIELDS.DATA]: data !== undefined ? data : null,
        [BIP_CONSTANTS.RESPONSE_FIELDS.TIMESTAMP]: getCurrentIsoTimestamp()
      };
    }

    /**
    * @param {string} action
    * @param {Error} error
    * @return {Object}
    */
    function buildErrorResponse(action, error) {
      const isAppError = error instanceof AppError;
      logError(`Action "${action}" failed: ${error.message}`, { stack: error.stack });
      return {
        [BIP_CONSTANTS.RESPONSE_FIELDS.SUCCESS]: false,
        [BIP_CONSTANTS.RESPONSE_FIELDS.ACTION]: action,
        [BIP_CONSTANTS.RESPONSE_FIELDS.ERROR]: error.message,
        [BIP_CONSTANTS.RESPONSE_FIELDS.ERROR_CODE]: isAppError ? error.errorCode : BIP_CONSTANTS.ERROR_CODES.INTERNAL_ERROR,
        [BIP_CONSTANTS.RESPONSE_FIELDS.TIMESTAMP]: getCurrentIsoTimestamp()
      };
    }

    /**
    * @param {Object} payload
    * @return {GoogleAppsScript.Content.TextOutput}
    */
    function toJsonTextOutput(payload) {
      return ContentService.createTextOutput(JSON.stringify(payload)).setMimeType(ContentService.MimeType.JSON);
    }

    /**
    * @param {function(number): *} operationFn
    * @param {number} maxRetries
    * @param {number} baseDelayMs
    * @return {*}
    * @throws {Error}
    */
    function retryWithBackoff(operationFn, maxRetries, baseDelayMs) {
      let attempt = 0;
      let lastError = null;
      while (attempt < maxRetries) {
        try {
          return operationFn(attempt);
        } catch (operationError) {
          lastError = operationError;
          attempt += 1;
          if (attempt >= maxRetries) {
            break;
          }
          const delayMs = baseDelayMs * Math.pow(2, attempt - 1);
          logWarn('Operation failed, retrying.', { attempt, delayMs, error: operationError.message });
          Utilities.sleep(delayMs);
        }
      }
      throw lastError;
    }

    /**
    * @param {function(): *} operationFn
    * @param {number} timeoutMs
    * @return {*}
    * @throws {AppError}
    */
    function withScriptLock(operationFn, timeoutMs) {
      const lock = LockService.getScriptLock();
      try {
        lock.waitLock(timeoutMs);
      } catch (lockError) {
        throw new AppError(

      }

        return operationFn();
      } finally {
        lock.releaseLock();
      }
    }

Config.gs

  /**
  * @fileoverview Configuration management for the BIP Automation system.
  *
  * Config.gs is the SOLE file permitted to interact directly with
  * PropertiesService, CacheService, and LockService. Every other file
  * (Services, Router, triggers) must obtain configuration values and
  * locks exclusively through the `ConfigService` object exposed here.
  *
  * ARCHITECTURAL POSITION (Layer 1 - Configuration)
  * Depends only on Constants.gs and Utilities.gs (BIP_UTILS).
  *
  * CACHING STRATEGY
  * All Script Properties are read once per cache window and stored as a
  * single JSON blob in CacheService.getScriptCache(), keyed by
  * CACHE_KEY_. This avoids repeated PropertiesService round-trips (which
  * are comparatively slow) across the many config getters called during
  * a single import/reminder run. Any write invalidates the cache
  * immediately so subsequent reads are never stale within the same
  * deployment.
  *
  * GLOBAL SCOPE NOTE
  * Exposes exactly ONE global symbol, `ConfigService`, built via an IIFE,
  * to avoid collisions in the shared Apps Script global namespace.
  */

  const ConfigService = Object.freeze((() => {

    const CACHE_KEY_ = 'BIP_CONFIG_CACHE_V1';
    const CACHE_TTL_SECONDS_ = 21600; // 6 hours - CacheService maximum.
    const LOCK_KEY_CONTEXT_ = 'BIP_SCRIPT_LOCK';

    const KEYS = BIP_CONSTANTS.SCRIPT_PROPERTY_KEYS;
    const DEFAULTS = BIP_CONSTANTS.DEFAULT_CONFIG_VALUES;
    const ERROR_CODES = BIP_CONSTANTS.ERROR_CODES;

    /** Script Properties keys that MUST be present with a non-empty value; no default exists. */
    const CRITICAL_KEYS_ = Object.freeze([
      KEYS.SPREADSHEET_ID,
      KEYS.DRIVE_ROOT_FOLDER_ID,
      KEYS.GEMINI_API_KEY
    ]);

    /**
    * Returns the PropertiesService script properties store.
    * @return {GoogleAppsScript.Properties.Properties}
    * @private
    */
    function getPropertiesStore_() {
      return PropertiesService.getScriptProperties();
    }

    /**
    * Returns the script-level CacheService cache.
    * @return {GoogleAppsScript.Cache.Cache}
    * @private
    */
    function getCacheStore_() {
      return CacheService.getScriptCache();
    }

    /**
    * Reads all script properties, transparently using the cache.
    * @return {Object<string, string>}
    * @private
    */
    function readAllProperties_() {
      const cache = getCacheStore_();
      let cached = null;
      try {
        cached = cache.get(CACHE_KEY_);
      } catch (err) {
        BIP_UTILS.logWarn('ConfigService', 'Cache read failed, falling back to PropertiesService.', { error: String(err) });
      }

      if (cached) {
        try {
          return JSON.parse(cached);
        } catch (parseErr) {
          BIP_UTILS.logWarn('ConfigService', 'Cached config JSON was corrupt, re-reading from PropertiesService.', { error: String(parseErr) });
        }
      }

      const properties = getPropertiesStore_().getProperties();

      try {
        cache.put(CACHE_KEY_, JSON.stringify(properties), CACHE_TTL_SECONDS_);
      } catch (cacheWriteErr) {
        BIP_UTILS.logWarn('ConfigService', 'Cache write failed.', { error: String(cacheWriteErr) });
      }

      return properties;
    }

    /**
    * Invalidates the cached properties blob. Must be called after any write.
    * @private
    */
    function invalidateCache_() {
      try {
        getCacheStore_().remove(CACHE_KEY_);
      } catch (err) {
        BIP_UTILS.logWarn('ConfigService', 'Cache invalidation failed.', { error: String(err) });
      }
    }

    /**
    * Retrieves a single raw property value, or null if unset/blank.
    * @param {string} key One of BIP_CONSTANTS.SCRIPT_PROPERTY_KEYS.
    * @return {?string}
    * @private
    */
    function getRawValue_(key) {
      const properties = readAllProperties_();
      const value = properties[key];
      return (value === undefined || value === null || value === '') ? null : value;
    }

    /**
    * Retrieves a non-critical property, falling back to DEFAULT_CONFIG_VALUES.
    * @param {string} key
    * @return {?string}
    * @private
    */
    function getWithDefault_(key) {
      const raw = getRawValue_(key);
      if (raw !== null) {
        return raw;
      }
      return Object.prototype.hasOwnProperty.call(DEFAULTS, key) ? String(DEFAULTS[key]) : null;
    }

    /**
    * Retrieves a critical property. Throws CONFIGURATION_ERROR if missing.
    * @param {string} key
    * @return {string}
    * @private
    */
    function requireCritical_(key) {
      const raw = getRawValue_(key);
      if (raw === null) {
        throw BIP_UTILS.createError(
          ERROR_CODES.CONFIGURATION_ERROR,
          `Missing required Script Property "${key}". Configure it before running BIP Automation.`
        );
      }
      return raw;
    }

    /**
    * Parses a property value expected to be an integer.
    * @param {string} key
    * @param {number} fallback
    * @return {number}
    * @private
    */
    function getIntWithDefault_(key, fallback) {
      const raw = getWithDefault_(key);
      const parsed = parseInt(raw, 10);
      return isNaN(parsed) ? fallback : parsed;
    }

    /**
    * Validates that a given hour is an integer in [0, 23].
    * @param {string} label
    * @param {*} value
    * @private
    */
    function assertValidHour_(label, value) {
      const parsed = Number(value);
      if (!Number.isInteger(parsed) || parsed < 0 || parsed > 23) {
        throw BIP_UTILS.createError(
          ERROR_CODES.VALIDATION_ERROR,
          `${label} must be an integer between 0 and 23. Received: ${value}`
        );
      }
    }

    /**
    * Validates that a given value is a positive integer.
    * @param {string} label
    * @param {*} value
    * @private
    */
    function assertPositiveInteger_(label, value) {
      const parsed = Number(value);
      if (!Number.isInteger(parsed) || parsed <= 0) {
        throw BIP_UTILS.createError(
          ERROR_CODES.VALIDATION_ERROR,
          `${label} must be a positive integer. Received: ${value}`
        );
      }
    }

    /**
    * Validates a non-empty string.
    * @param {string} label
    * @param {*} value
    * @private
    */
    function assertNonEmptyString_(label, value) {
      if (typeof value !== 'string' || value.trim() === '') {
        throw BIP_UTILS.createError(
          ERROR_CODES.VALIDATION_ERROR,
          `${label} must be a non-empty string.`
        );
      }
    }

    /**
    * Masks a secret value for display purposes, keeping only the last 4 characters.
    * @param {?string} value
    * @return {string}
    * @private
    */
    function maskSecret_(value) {
      if (!value) {
        return '';
      }
      if (value.length <= 4) {
        return '*'.repeat(value.length);
      }
      return `${'*'.repeat(value.length - 4)}${value.slice(-4)}`;
    }

    return {

      /**
      * Returns the target Google Spreadsheet ID. Critical property.
      * @return {string}
      */
      getSpreadsheetId() {
        return requireCritical_(KEYS.SPREADSHEET_ID);
      },

      /**
      * Returns the Google Drive root folder ID under which per-BIP
      * subfolders are created. Critical property.
      * @return {string}
      */
      getDriveRootFolderId() {
        return requireCritical_(KEYS.DRIVE_ROOT_FOLDER_ID);
      },

      /**
      * Returns the Gemini Generative Language API key. Critical property.
      * @return {string}
      */
      getGeminiApiKey() {
        return requireCritical_(KEYS.GEMINI_API_KEY);
      },

      /**
      * Returns the Gemini model name, falling back to DEFAULT_CONFIG_VALUES.GEMINI_MODEL.
      * @return {string}
      */
      getGeminiModel() {
        return getWithDefault_(KEYS.GEMINI_MODEL);
      },

      /**
      * Returns the Gmail label name applied to processed messages.
      * @return {string}
      */
      getGmailLabelName() {
        return getWithDefault_(KEYS.GMAIL_LABEL_NAME);
      },

      /**
      * Returns the Gmail search query used to find BIP invitation emails.
      * @return {string}
      */
      getGmailSearchQuery() {
        return getWithDefault_(KEYS.GMAIL_SEARCH_QUERY);
      },

      /**
      * Returns the number of days between automated reminder emails.
      * @return {number}
      */
      getReminderIntervalDays() {
        return getIntWithDefault_(KEYS.REMINDER_INTERVAL_DAYS, DEFAULTS.REMINDER_INTERVAL_DAYS);
      },

      /**
      * Returns the IANA timezone string used for date formatting and triggers.
      * @return {string}
      */
      getTimezone() {
        return getWithDefault_(KEYS.TIMEZONE);
      },

      /**
      * Returns the hour of day (0-23) at which the scheduled import runs.
      * @return {number}
      */
      getImportTriggerHour() {
        return getIntWithDefault_(KEYS.IMPORT_TRIGGER_HOUR, DEFAULTS.IMPORT_TRIGGER_HOUR);
      },

      /**
      * Returns the hour of day (0-23) at which scheduled reminders run.
      * @return {number}
      */
      getReminderTriggerHour() {
        return getIntWithDefault_(KEYS.REMINDER_TRIGGER_HOUR, DEFAULTS.REMINDER_TRIGGER_HOUR);
      },

      /**
      * Returns the administrator alert email address, or null if unset.
      * This property is optional and has no default.
      * @return {?string}

      getAdminAlertEmail() {
        return getRawValue_(KEYS.ADMIN_ALERT_EMAIL);
      },

      /**
      * Returns the display name used as the sender name for notification emails.
      * @return {string}
      */
      getNotificationSenderName() {
        return getWithDefault_(KEYS.NOTIFICATION_SENDER_NAME);
      },

      /**
      * Returns all configuration values as a plain object suitable for the
      * Settings web app UI. The Gemini API key is masked; callers must use
      * updateSettings() to change it, never read it back in full.
      * @return {Object<string, string>}
      */
      getAllSettings() {
        const properties = readAllProperties_();
        const result = {};
        Object.keys(KEYS).forEach((constName) => {
          const propertyKey = KEYS[constName];
          const rawValue = properties[propertyKey];
          const effectiveValue = (rawValue === undefined || rawValue === null || rawValue === '')
            ? (Object.prototype.hasOwnProperty.call(DEFAULTS, propertyKey) ? String(DEFAULTS[propertyKey]) : '')
            : rawValue;
          result[propertyKey] = (propertyKey === KEYS.GEMINI_API_KEY) ? maskSecret_(effectiveValue) : effectiveValue;
        });
        return result;
      },

      /**
      * Validates and persists one or more configuration values in a single
      * batch write. Only recognized keys (BIP_CONSTANTS.SCRIPT_PROPERTY_KEYS)
      * are accepted; unknown keys throw a VALIDATION_ERROR. A blank/omitted
      * GEMINI_API_KEY is ignored (does not overwrite the existing secret),
      * so the settings UI never has to redisplay or resubmit it.
      * @param {Object<string, *>} settings Map of SCRIPT_PROPERTY_KEYS value -> new value.
      * @return {void}
      */
      updateSettings(settings) {
        if (!settings || typeof settings !== 'object') {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'updateSettings requires a settings object.');
        }

        const validKeys = Object.values(KEYS);
        const toWrite = {};

        Object.keys(settings).forEach((key) => {
          if (validKeys.indexOf(key) === -1) {
            throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, `Unknown configuration key: "${key}".`);
          }
        });

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.SPREADSHEET_ID)) {
          assertNonEmptyString_('SPREADSHEET_ID', settings[KEYS.SPREADSHEET_ID]);
          toWrite[KEYS.SPREADSHEET_ID] = settings[KEYS.SPREADSHEET_ID].trim();
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.DRIVE_ROOT_FOLDER_ID)) {
          assertNonEmptyString_('DRIVE_ROOT_FOLDER_ID', settings[KEYS.DRIVE_ROOT_FOLDER_ID]);
          toWrite[KEYS.DRIVE_ROOT_FOLDER_ID] = settings[KEYS.DRIVE_ROOT_FOLDER_ID].trim();
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.GEMINI_API_KEY)) {
          const apiKey = settings[KEYS.GEMINI_API_KEY];
          if (typeof apiKey === 'string' && apiKey.trim() !== '') {
            toWrite[KEYS.GEMINI_API_KEY] = apiKey.trim();
          }
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.GEMINI_MODEL)) {
          assertNonEmptyString_('GEMINI_MODEL', settings[KEYS.GEMINI_MODEL]);
          toWrite[KEYS.GEMINI_MODEL] = settings[KEYS.GEMINI_MODEL].trim();
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.GMAIL_LABEL_NAME)) {
          assertNonEmptyString_('GMAIL_LABEL_NAME', settings[KEYS.GMAIL_LABEL_NAME]);
          toWrite[KEYS.GMAIL_LABEL_NAME] = settings[KEYS.GMAIL_LABEL_NAME].trim();
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.GMAIL_SEARCH_QUERY)) {
          assertNonEmptyString_('GMAIL_SEARCH_QUERY', settings[KEYS.GMAIL_SEARCH_QUERY]);
          toWrite[KEYS.GMAIL_SEARCH_QUERY] = settings[KEYS.GMAIL_SEARCH_QUERY].trim();
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.REMINDER_INTERVAL_DAYS)) {
          assertPositiveInteger_('REMINDER_INTERVAL_DAYS', settings[KEYS.REMINDER_INTERVAL_DAYS]);
          toWrite[KEYS.REMINDER_INTERVAL_DAYS] = String(parseInt(settings[KEYS.REMINDER_INTERVAL_DAYS], 10));
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.TIMEZONE)) {
          assertNonEmptyString_('TIMEZONE', settings[KEYS.TIMEZONE]);
          toWrite[KEYS.TIMEZONE] = settings[KEYS.TIMEZONE].trim();
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.IMPORT_TRIGGER_HOUR)) {
          assertValidHour_('IMPORT_TRIGGER_HOUR', settings[KEYS.IMPORT_TRIGGER_HOUR]);
          toWrite[KEYS.IMPORT_TRIGGER_HOUR] = String(parseInt(settings[KEYS.IMPORT_TRIGGER_HOUR], 10));
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.REMINDER_TRIGGER_HOUR)) {
          assertValidHour_('REMINDER_TRIGGER_HOUR', settings[KEYS.REMINDER_TRIGGER_HOUR]);
          toWrite[KEYS.REMINDER_TRIGGER_HOUR] = String(parseInt(settings[KEYS.REMINDER_TRIGGER_HOUR], 10));
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.ADMIN_ALERT_EMAIL)) {
          const email = settings[KEYS.ADMIN_ALERT_EMAIL];
          if (typeof email === 'string' && email.trim() !== '') {
            if (!BIP_UTILS.isValidEmail(email.trim())) {
              throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, `ADMIN_ALERT_EMAIL is not a valid email address: "${email}".`);
            }
            toWrite[KEYS.ADMIN_ALERT_EMAIL] = email.trim();
          } else {
            toWrite[KEYS.ADMIN_ALERT_EMAIL] = '';
          }
        }

        if (Object.prototype.hasOwnProperty.call(settings, KEYS.NOTIFICATION_SENDER_NAME)) {
          assertNonEmptyString_('NOTIFICATION_SENDER_NAME', settings[KEYS.NOTIFICATION_SENDER_NAME]);
          toWrite[KEYS.NOTIFICATION_SENDER_NAME] = settings[KEYS.NOTIFICATION_SENDER_NAME].trim();
        }

        if (Object.keys(toWrite).length === 0) {
          return;
        }

        getPropertiesStore_().setProperties(toWrite, false);
        invalidateCache_();
        BIP_UTILS.logInfo('ConfigService', 'Configuration updated.', { keys: Object.keys(toWrite) });
      },

      /**
      * Acquires a script-wide exclusive lock, blocking up to
      * BIP_CONSTANTS.TIME.SCRIPT_LOCK_TIMEOUT_MS milliseconds.
      * @return {GoogleAppsScript.Lock.Lock} The acquired lock; caller MUST release it.
      */
      acquireLock() {
        const lock = LockService.getScriptLock();
        const timeoutMs = BIP_CONSTANTS.TIME.SCRIPT_LOCK_TIMEOUT_MS;
        const acquired = lock.tryLock(timeoutMs);
        if (!acquired) {
          throw BIP_UTILS.createError(
            ERROR_CODES.INTERNAL_ERROR,
            `Could not acquire ${LOCK_KEY_CONTEXT_} within ${timeoutMs}ms; another execution is likely still running.`
          );
        }
        return lock;
      },

      /**
      * Releases a previously acquired lock. Safe to call even if the lock
      * is already released; failures are logged, not thrown.
      * @param {GoogleAppsScript.Lock.Lock} lock
      * @return {void}
      */
      releaseLock(lock) {
        if (!lock) {
          return;
        }
        try {
          lock.releaseLock();
        } catch (err) {
          BIP_UTILS.logWarn('ConfigService', 'Failed to release script lock.', { error: String(err) });
        }
      },

      /**
      * Runs the given callback while holding the exclusive script lock,
      * guaranteeing the lock is released afterwards even if the callback
      * throws. Use this to wrap scheduled import/reminder entry points so
      * overlapping trigger executions cannot run concurrently.
      * @param {function(): *} callback
      * @return {*} The callback's return value.
      */
      runExclusive(callback) {
        const lock = this.acquireLock();
        try {
          return callback();
        } finally {
          this.releaseLock(lock);
        }
      }
    };

  })());

      if (lastColumn < 1) {
        throw BIP_UTILS.createError(
          ERROR_CODES.SPREADSHEET_ERROR,
          `Sheet "${sheet.getName()}" has no header row.`
        );
      }
      const headerRow = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
      const map = {};
      headerRow.forEach((headerText, idx) => {
        const trimmed = String(headerText).trim();
        if (trimmed !== '') {
          map[trimmed] = idx + 1;
        }
      });
      return map;
    }

    /**
    * Builds (and caches) the BIP_FIELD_KEYS -> column-index map for the "Main" sheet.
    * @return {Object<string, number>}
    * @private
    */
    function getMainFieldColumnMap_() {
      if (cachedMainHeaderMap_) {
        return cachedMainHeaderMap_;
      }
      const headerIndexMap = buildHeaderIndexMap_(getMainSheet_());
      const fieldColumnMap = {};
      const missingHeaders = [];

      Object.keys(MAIN_SHEET_COLUMN_HEADERS_).forEach((fieldKey) => {
        const expectedHeader = MAIN_SHEET_COLUMN_HEADERS_[fieldKey];
        const columnIndex = headerIndexMap[expectedHeader];
        if (columnIndex === undefined) {
          missingHeaders.push(expectedHeader);
        } else {
          fieldColumnMap[fieldKey] = columnIndex;
        }
      });

      if (missingHeaders.length > 0) {
        throw BIP_UTILS.createError(
          ERROR_CODES.SPREADSHEET_ERROR,
          `"${SHEETS.MAIN}" sheet is missing expected header column(s): ${missingHeaders.join(', ')}.`
        );
      }

      cachedMainHeaderMap_ = fieldColumnMap;
      return cachedMainHeaderMap_;
    }

    /**
    * Normalizes a raw sheet cell value for a given BIP field key into the
    * representation used throughout the rest of the codebase (plain
    * strings/numbers, ISO-formatted dates).
    * @param {string} fieldKey
    * @param {*} rawValue
    * @return {*}
    * @private
    */
    function normalizeCellValue_(fieldKey, rawValue) {
      if (rawValue instanceof Date) {
        const timezone = ConfigService.getTimezone();
        if (DATE_ONLY_FIELDS_.indexOf(fieldKey) !== -1) {
          return Utilities.formatDate(rawValue, timezone, 'yyyy-MM-dd');
        }
        if (DATE_TIME_FIELDS_.indexOf(fieldKey) !== -1) {
          return Utilities.formatDate(rawValue, timezone, "yyyy-MM-dd'T'HH:mm:ss");
        }
        return Utilities.formatDate(rawValue, timezone, 'yyyy-MM-dd');
      }
      if (fieldKey === FIELD_KEYS.REMINDER_COUNT) {
        const parsed = parseInt(rawValue, 10);
        return isNaN(parsed) ? 0 : parsed;
      }
      if (rawValue === null || rawValue === undefined) {
        return '';
      }
      return rawValue;
    }

    /**
    * Converts a full "Main" sheet row into a BIP record object keyed by
    * BIP_FIELD_KEYS, plus an internal, non-persisted `_rowNumber`.
    * @param {Array<*>} rowValues Full row values (1 per sheet column).
    * @param {number} rowNumber 1-based sheet row number.
    * @param {Object<string, number>} fieldColumnMap
    * @return {Object}
    * @private
    */
    function rowToRecord_(rowValues, rowNumber, fieldColumnMap) {
      const record = { _rowNumber: rowNumber };
      Object.keys(fieldColumnMap).forEach((fieldKey) => {
        const columnIndex = fieldColumnMap[fieldKey];
        record[fieldKey] = normalizeCellValue_(fieldKey, rowValues[columnIndex - 1]);
      });
      return record;
    }

    /**
    * Validates a candidate BIP record before it is appended, ensuring all
    * mandatory identifying fields are present and internally consistent.
    * @param {Object} record
    * @private
    */
    function assertValidNewRecord_(record) {
      if (!record || typeof record !== 'object') {
        throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'BIP record must be a non-null object.');
      }
      if (!record[FIELD_KEYS.GMAIL_MESSAGE_ID] || String(record[FIELD_KEYS.GMAIL_MESSAGE_ID]).trim() === '') {
        throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, `Field "${FIELD_KEYS.GMAIL_MESSAGE_ID}" is required on every BIP record.`);
      }
      if (record[FIELD_KEYS.ACADEMIC_LEVEL] && VALID_ACADEMIC_LEVELS.indexOf(record[FIELD_KEYS.ACADEMIC_LEVEL]) === -1) {
        throw BIP_UTILS.createError(
          ERROR_CODES.VALIDATION_ERROR,
          `Field "${FIELD_KEYS.ACADEMIC_LEVEL}" must be one of: ${VALID_ACADEMIC_LEVELS.join(', ')}. Received: "${record[FIELD_KEYS.ACADEMIC_LEVEL]}".`
        );
      }
      [FIELD_KEYS.START_DATE, FIELD_KEYS.END_DATE, FIELD_KEYS.APPLICATION_DEADLINE].forEach((dateField) => {
        const value = record[dateField];
        if (value && typeof value === 'string' && !BIP_UTILS.isValidIsoDate(value)) {
          throw BIP_UTILS.createError(
            ERROR_CODES.VALIDATION_ERROR,
            `Field "${dateField}" must be an ISO date (yyyy-MM-dd) string. Received: "${value}".`
          );
        }
      });
    }

    /**
    * Locates the sheet row for a given Gmail message ID.
    * @param {string} gmailMessageId
    * @return {?{record: Object, rowNumber: number}}
    * @private
    */
    function findRowByGmailMessageId_(gmailMessageId) {
      const sheet = getMainSheet_();
      const fieldColumnMap = getMainFieldColumnMap_();
      const lastRow = sheet.getLastRow();
      if (lastRow < 2) {
        return null;
      }
      const gmailIdColumn = fieldColumnMap[FIELD_KEYS.GMAIL_MESSAGE_ID];
      const idValues = sheet.getRange(2, gmailIdColumn, lastRow - 1, 1).getValues();
      for (let i = 0; i < idValues.length; i++) {
        if (String(idValues[i][0]).trim() === String(gmailMessageId).trim()) {
          const rowNumber = i + 2;
          const lastColumn = sheet.getLastColumn();
          const rowValues = sheet.getRange(rowNumber, 1, 1, lastColumn).getValues()[0];
          return { record: rowToRecord_(rowValues, rowNumber, fieldColumnMap), rowNumber };
        }
      }
      return null;
    }

    return {

      /**
      * Appends a new BIP record to the "Main" sheet. Rejects duplicates
      * (matched by gmailMessageId) with a DUPLICATE_ENTRY error so the
      * import pipeline can safely retry/re-run without creating repeats.
      * @param {Object} record Plain object keyed by BIP_CONSTANTS.BIP_FIELD_KEYS values.
      * @return {Object} The persisted record, including its `_rowNumber`.
      */
      appendBipRecord(record) {
        assertValidNewRecord_(record);

        const existing = findRowByGmailMessageId_(record[FIELD_KEYS.GMAIL_MESSAGE_ID]);
        if (existing) {
          throw BIP_UTILS.createError(
            ERROR_CODES.DUPLICATE_ENTRY,
            `A BIP record for Gmail message "${record[FIELD_KEYS.GMAIL_MESSAGE_ID]}" already exists at row ${existing.rowNumber}.`
          );
        }

        const sheet = getMainSheet_();
        const fieldColumnMap = getMainFieldColumnMap_();
        const lastColumn = sheet.getLastColumn();
        const newRowValues = new Array(lastColumn).fill('');

        Object.keys(fieldColumnMap).forEach((fieldKey) => {
          if (Object.prototype.hasOwnProperty.call(record, fieldKey) && record[fieldKey] !== undefined) {
            newRowValues[fieldColumnMap[fieldKey] - 1] = record[fieldKey];
          } else if (fieldKey === FIELD_KEYS.DECISION_STATUS) {
            newRowValues[fieldColumnMap[fieldKey] - 1] = DECISION_STATUS.PENDING;
          } else if (fieldKey === FIELD_KEYS.REMINDER_COUNT) {
            newRowValues[fieldColumnMap[fieldKey] - 1] = 0;
          }
        });

        const newRowNumber = sheet.getLastRow() + 1;
        sheet.getRange(newRowNumber, 1, 1, lastColumn).setValues([newRowValues]);

        BIP_UTILS.logInfo('SpreadsheetService', 'Appended new BIP record.', {
          gmailMessageId: record[FIELD_KEYS.GMAIL_MESSAGE_ID],
          rowNumber: newRowNumber
        });

        return rowToRecord_(newRowValues, newRowNumber, fieldColumnMap);
      },

      /**
      * Finds a single BIP record by its Gmail message ID.
      * @param {string} gmailMessageId
      * @return {?Object} The record (with `_rowNumber`), or null if not found.
      */
      findRecordByGmailMessageId(gmailMessageId) {
        if (!gmailMessageId) {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'gmailMessageId is required.');
        }
        const found = findRowByGmailMessageId_(gmailMessageId);
        return found ? found.record : null;
      },

      /**
      * Returns every BIP record currently stored in the "Main" sheet.
      * @return {Object[]} Array of records, each including `_rowNumber`.
      */
      getAllBipRecords() {
        const sheet = getMainSheet_();
        const fieldColumnMap = getMainFieldColumnMap_();
        const lastRow = sheet.getLastRow();
        const lastColumn = sheet.getLastColumn();
        if (lastRow < 2) {
          return [];
        }
        const values = sheet.getRange(2, 1, lastRow - 1, lastColumn).getValues();
        return values.map((rowValues, idx) => rowToRecord_(rowValues, idx + 2, fieldColumnMap));
      },

      /**
      * Updates one or more fields of an existing BIP record, identified by
      * its Gmail message ID. Only keys present in BIP_CONSTANTS.BIP_FIELD_KEYS
      * are accepted, and gmailMessageId itself is immutable.
      * @param {string} gmailMessageId
      * @param {Object<string, *>} fieldsToUpdate
      * @return {Object} The updated record.
      */
      updateRecordFields(gmailMessageId, fieldsToUpdate) {
        if (!fieldsToUpdate || typeof fieldsToUpdate !== 'object' || Object.keys(fieldsToUpdate).length === 0) {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'fieldsToUpdate must be a non-empty object.');
        }

        const found = findRowByGmailMessageId_(gmailMessageId);
        if (!found) {
          throw BIP_UTILS.createError(
            ERROR_CODES.NOT_FOUND,
            `No BIP record found for Gmail message "${gmailMessageId}".`
          );
        }

        const fieldColumnMap = getMainFieldColumnMap_();
        const validFieldKeys = Object.values(FIELD_KEYS);
        const sheet = getMainSheet_();

        Object.keys(fieldsToUpdate).forEach((fieldKey) => {
          if (validFieldKeys.indexOf(fieldKey) === -1) {
            throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, `Unknown BIP field key: "${fieldKey}".`);
          }
          if (fieldKey === FIELD_KEYS.GMAIL_MESSAGE_ID) {
            throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, `Field "${FIELD_KEYS.GMAIL_MESSAGE_ID}" is immutable and cannot be updated.`);
          }
          if (fieldKey === FIELD_KEYS.DECISION_STATUS) {
            const validStatuses = Object.values(DECISION_STATUS);
            if (validStatuses.indexOf(fieldsToUpdate[fieldKey]) === -1) {
              throw BIP_UTILS.createError(
                ERROR_CODES.VALIDATION_ERROR,
                `Field "${FIELD_KEYS.DECISION_STATUS}" must be one of: ${validStatuses.map((s) => `"${s}"`).join(', ')}.`
              );
            }
          }
          sheet.getRange(found.rowNumber, fieldColumnMap[fieldKey]).setValue(fieldsToUpdate[fieldKey]);
        });

        BIP_UTILS.logInfo('SpreadsheetService', 'Updated BIP record fields.', {
          gmailMessageId,
          rowNumber: found.rowNumber,
          fields: Object.keys(fieldsToUpdate)
        });

        const lastColumn = sheet.getLastColumn();
        const refreshedRow = sheet.getRange(found.rowNumber, 1, 1, lastColumn).getValues()[0];
        return rowToRecord_(refreshedRow, found.rowNumber, fieldColumnMap);
      },

      /**
      * Convenience wrapper: sets a BIP record's decision status.
      * @param {string} gmailMessageId
      * @param {string} newStatus One of BIP_CONSTANTS.DECISION_STATUS values.
      * @return {Object} The updated record.
      */
      updateDecisionStatus(gmailMessageId, newStatus) {
        const update = {};
        update[FIELD_KEYS.DECISION_STATUS] = newStatus;
        return this.updateRecordFields(gmailMessageId, update);
      },

      /**
      * Convenience wrapper: records that a reminder email was just sent,
      * stamping the current timestamp and incrementing the reminder count.
      * @param {string} gmailMessageId
      * @param {string} sentAtIso ISO 8601 timestamp of the send.
      * @return {Object} The updated record.
      */
      recordReminderSent(gmailMessageId, sentAtIso) {
        const existing = this.findRecordByGmailMessageId(gmailMessageId);
        if (!existing) {
          throw BIP_UTILS.createError(ERROR_CODES.NOT_FOUND, `No BIP record found for Gmail message "${gmailMessageId}".`);
        }
        const currentCount = parseInt(existing[FIELD_KEYS.REMINDER_COUNT], 10) || 0;
        const update = {};
        update[FIELD_KEYS.LAST_REMINDER_SENT_AT] = sentAtIso;
        update[FIELD_KEYS.REMINDER_COUNT] = currentCount + 1;
        return this.updateRecordFields(gmailMessageId, update);
      },

      /**
      * Convenience wrapper: stores the Google Drive folder URL created for a record.
      * @param {string} gmailMessageId
      * @param {string} driveFolderUrl
      * @return {Object} The updated record.
      */
      setDriveFolderUrl(gmailMessageId, driveFolderUrl) {
        if (!driveFolderUrl || String(driveFolderUrl).trim() === '') {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'driveFolderUrl must be a non-empty string.');
        }
        const update = {};
        update[FIELD_KEYS.DRIVE_FOLDER_URL] = driveFolderUrl;
        return this.updateRecordFields(gmailMessageId, update);
      },

      /**
      * Returns every BIP record eligible for a reminder email: decision
      * status is still pending, and the configured interval has elapsed
      * since the last reminder (or since import, if none was sent yet).
      * @param {number} intervalDays Minimum number of days between reminders.
      * @return {Object[]} Eligible records.
      */
      getRecordsPendingReminder(intervalDays) {
        if (!Number.isInteger(intervalDays) || intervalDays <= 0) {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'intervalDays must be a positive integer.');
        }
        const intervalMs = intervalDays * TIME.MILLISECONDS_PER_DAY;
        const now = Date.now();

        return this.getAllBipRecords().filter((rec) => {
          if (rec[FIELD_KEYS.DECISION_STATUS] !== DECISION_STATUS.PENDING) {
            return false;
          }
          const referenceValue = rec[FIELD_KEYS.LAST_REMINDER_SENT_AT] || rec[FIELD_KEYS.IMPORTED_AT];
          if (!referenceValue) {
            return true;
          }
          const referenceTime = new Date(referenceValue).getTime();
          if (isNaN(referenceTime)) {
            return true;
          }
          return (now - referenceTime) >= intervalMs;
        });
      },

      /**
      * Computes summary statistics across all BIP records, for the
      * "statistics" Router action.
      * @return {{totalRecords: number, byDecisionStatus: Object<string, number>, byAcademicLevel: Object<string, number>, upcomingDeadlines: number}}
      */
      getStatistics() {
        const records = this.getAllBipRecords();
        const byDecisionStatus = {};
        Object.values(DECISION_STATUS).forEach((status) => {
          const label = status === '' ? 'pending' : status.toLowerCase();
          byDecisionStatus[label] = 0;
        });
        const byAcademicLevel = {};
        VALID_ACADEMIC_LEVELS.forEach((level) => {
          byAcademicLevel[level] = 0;
        });

        let upcomingDeadlines = 0;
        const now = Date.now();

        records.forEach((rec) => {
          const statusLabel = rec[FIELD_KEYS.DECISION_STATUS] === '' ? 'pending' : String(rec[FIELD_KEYS.DECISION_STATUS]).toLowerCase();
          byDecisionStatus[statusLabel] = (byDecisionStatus[statusLabel] || 0) + 1;

          const level = rec[FIELD_KEYS.ACADEMIC_LEVEL];
          if (level && Object.prototype.hasOwnProperty.call(byAcademicLevel, level)) {
            byAcademicLevel[level] += 1;
          }

          const deadline = rec[FIELD_KEYS.APPLICATION_DEADLINE];
          if (deadline) {
            const deadlineTime = new Date(deadline).getTime();
            if (!isNaN(deadlineTime) && deadlineTime >= now) {
              upcomingDeadlines += 1;
            }
          }
        });

        return {
          totalRecords: records.length,
          byDecisionStatus,
          byAcademicLevel,
          upcomingDeadlines
        };
      },

      /**
      * Returns every row of the "Faculty Coordinators" sheet as plain objects.
      * @return {Object[]} Array keyed by BIP_CONSTANTS.COORDINATOR_FIELD_KEYS.
      */
      getFacultyCoordinators() {
        const sheet = getFacultyCoordinatorsSheet_();
        const fieldColumnMap = getFacultyColumnMap_();
        const lastRow = sheet.getLastRow();
        const lastColumn = sheet.getLastColumn();
        if (lastRow < 2) {
          return [];
        }
        const values = sheet.getRange(2, 1, lastRow - 1, lastColumn).getValues();
        return values
          .filter((row) => row[fieldColumnMap[COORD_KEYS.FACULTY_NAME] - 1] !== '')
          .map((row) => ({
            [COORD_KEYS.FACULTY_NAME]: String(row[fieldColumnMap[COORD_KEYS.FACULTY_NAME] - 1]).trim(),
            [COORD_KEYS.COORDINATOR_FULL_NAME]: String(row[fieldColumnMap[COORD_KEYS.COORDINATOR_FULL_NAME] - 1]).trim(),
            [COORD_KEYS.COORDINATOR_EMAIL]: String(row[fieldColumnMap[COORD_KEYS.COORDINATOR_EMAIL] - 1]).trim()
          }));
      },

      /**
      * Returns the coordinator(s) registered for a given faculty name
      * (case-insensitive, trimmed match).
      * @param {string} facultyName
      * @return {Object[]} Matching coordinator records.
      */
      getFacultyCoordinatorsByFaculty(facultyName) {
        if (!facultyName || String(facultyName).trim() === '') {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'facultyName is required.');
        }
        const normalizedTarget = String(facultyName).trim().toLowerCase();
        return this.getFacultyCoordinators().filter(
          (coordinator) => coordinator[COORD_KEYS.FACULTY_NAME].toLowerCase() === normalizedTarget
        );
      },

      /**
      * Returns a single email template by its "Template" column value.
      * @param {string} templateName
      * @return {{subject: string, body: string}} The template's subject and HTML body.
      */
      getEmailTemplate(templateName) {
        if (!templateName || String(templateName).trim() === '') {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'templateName is required.');
        }
        const sheet = getEmailTemplatesSheet_();
        const fieldColumnMap = getTemplateColumnMap_();
        const lastRow = sheet.getLastRow();
        const lastColumn = sheet.getLastColumn();
        if (lastRow >= 2) {
          const values = sheet.getRange(2, 1, lastRow - 1, lastColumn).getValues();
          const normalizedTarget = String(templateName).trim().toLowerCase();
          for (let i = 0; i < values.length; i++) {
            const row = values[i];
            const rowTemplateName = String(row[fieldColumnMap.template - 1]).trim();
            if (rowTemplateName.toLowerCase() === normalizedTarget) {
              return {
                subject: String(row[fieldColumnMap.subject - 1]),
                body: String(row[fieldColumnMap.body - 1])
              };
            }
          }
        }
        throw BIP_UTILS.createError(
          ERROR_CODES.NOT_FOUND,
          `Email template "${templateName}" was not found in the "${SHEETS.EMAIL_TEMPLATES}" sheet.`
        );
      },

      /**
      * Returns every email template defined in the "Email Templates" sheet.
      * @return {Array<{template: string, subject: string, body: string}>}
      */
      getAllEmailTemplates() {
        const sheet = getEmailTemplatesSheet_();
        const fieldColumnMap = getTemplateColumnMap_();
        const lastRow = sheet.getLastRow();
        const lastColumn = sheet.getLastColumn();
        if (lastRow < 2) {
          return [];
        }
        const values = sheet.getRange(2, 1, lastRow - 1, lastColumn).getValues();
        return values
          .filter((row) => String(row[fieldColumnMap.template - 1]).trim() !== '')
          .map((row) => ({
            template: String(row[fieldColumnMap.template - 1]).trim(),
            subject: String(row[fieldColumnMap.subject - 1]),
            body: String(row[fieldColumnMap.body - 1])
          }));
      }
    };

DriveService.gs

  /**
  * @fileoverview Google Drive persistence layer for the BIP Automation system.
  *
  * DriveService.gs is the SOLE file permitted to interact directly with
  * DriveApp. Every other file that needs to create BIP record folders or
  * store source-email artifacts must go exclusively through the
  * `DriveService` object exposed here.
  *
  * ARCHITECTURAL POSITION (Layer 2 - Persistence)
  * Depends on Constants.gs, Utilities.gs (BIP_UTILS) and Config.gs
  * (ConfigService) for the Drive root folder ID. Never acquires
  * PropertiesService/CacheService/LockService directly.
  *
  * FOLDER STRUCTURE
  * One subfolder per BIP record is created directly under the
  * configured root folder (BIP_CONSTANTS.SCRIPT_PROPERTY_KEYS -
  * DRIVE_ROOT_FOLDER_ID, resolved via ConfigService). Folder creation is
  * idempotent: calling createBipRecordFolder() twice with the same
  * university/title reuses the existing folder instead of creating a
  * duplicate, which makes the import pipeline safe to retry.
  *
  * GLOBAL SCOPE NOTE
  * Exposes exactly ONE global symbol, `DriveService`, built via an IIFE,
  * to avoid collisions in the shared Apps Script global namespace.
  */

  const DriveService = Object.freeze((() => {

    const ERROR_CODES = BIP_CONSTANTS.ERROR_CODES;
    const DRIVE_DEFAULTS = BIP_CONSTANTS.DRIVE_DEFAULTS;

    const MAX_FOLDER_NAME_LENGTH_ = 180;
    const FOLDER_NAME_SEPARATOR_ = ' - ';

    let cachedRootFolder_ = null;

    /**
    * Opens (and caches for this execution) the configured Drive root folder.
    * @return {GoogleAppsScript.Drive.Folder}
    * @private
    */
    function getRootFolder_() {
      if (cachedRootFolder_) {
        return cachedRootFolder_;
      }
      const rootFolderId = ConfigService.getDriveRootFolderId();
      try {
        cachedRootFolder_ = DriveApp.getFolderById(rootFolderId);
      } catch (err) {
        throw BIP_UTILS.createError(
          ERROR_CODES.DRIVE_ERROR,
          `Unable to open Drive root folder with ID "${rootFolderId}": ${err.message || err}`
        );
      }
      return cachedRootFolder_;
    }

    /**
    * Removes characters that are problematic or confusing in Drive folder
    * names, collapses whitespace, and enforces a maximum length.
    * @param {*} rawName
    * @param {string} fallbackName Used if the sanitized result is empty.
    * @return {string}
    * @private
    */
    function sanitizeName_(rawName, fallbackName) {
      let name = (rawName === null || rawName === undefined) ? '' : String(rawName);
      name = name.replace(/[\/\\]/g, '-');
      name = name.replace(/[\r\n\t]+/g, ' ');
      name = name.replace(/\s+/g, ' ').trim();
      if (name.length > MAX_FOLDER_NAME_LENGTH_) {
        name = name.substring(0, MAX_FOLDER_NAME_LENGTH_).trim();
      }
      return name === '' ? fallbackName : name;
    }

    /**
    * Builds the canonical folder name for a BIP record.
    * @param {string} universityName
    * @param {string} bipTitle
    * @return {string}
    * @private
    */
    function buildBipFolderName_(universityName, bipTitle) {
      const university = sanitizeName_(universityName, DRIVE_DEFAULTS.UNKNOWN_UNIVERSITY_FOLDER_NAME);
      const title = sanitizeName_(bipTitle, DRIVE_DEFAULTS.UNTITLED_BIP_FOLDER_NAME);
      const combined = `${university}${FOLDER_NAME_SEPARATOR_}${title}`;
      return combined.length > MAX_FOLDER_NAME_LENGTH_
        ? combined.substring(0, MAX_FOLDER_NAME_LENGTH_).trim()
        : combined;
    }

    /**
    * Finds the first direct subfolder of `parentFolder` with an exact
    * name match, or null if none exists.
    * @param {GoogleAppsScript.Drive.Folder} parentFolder
    * @param {string} folderName
    * @return {?GoogleAppsScript.Drive.Folder}
    * @private
    */
    function findFolderByName_(parentFolder, folderName) {
      const iterator = parentFolder.getFoldersByName(folderName);
      return iterator.hasNext() ? iterator.next() : null;
    }

    /**
    * Returns the existing subfolder matching `folderName`, or creates it.
    * @param {GoogleAppsScript.Drive.Folder} parentFolder
    * @param {string} folderName
    * @return {GoogleAppsScript.Drive.Folder}
    * @private
    */
    function getOrCreateFolder_(parentFolder, folderName) {
      try {
        const existing = findFolderByName_(parentFolder, folderName);
        if (existing) {
          return existing;
        }
        return parentFolder.createFolder(folderName);
      } catch (err) {
        throw BIP_UTILS.createError(
          ERROR_CODES.DRIVE_ERROR,
          `Unable to get or create folder "${folderName}": ${err.message || err}`
        );
      }
    }

    /**
    * Wraps a Drive folder handle into the plain object shape returned to callers.
    * @param {GoogleAppsScript.Drive.Folder} folder
    * @return {{folderId: string, folderUrl: string, folderName: string}}
    * @private
    */
    function toFolderInfo_(folder) {
      return {
        folderId: folder.getId(),
        folderUrl: folder.getUrl(),
        folderName: folder.getName()
      };
    }

    /**
    * Resolves a Drive folder by ID, throwing NOT_FOUND if it cannot be
    * accessed.
    * @param {string} folderId
    * @return {GoogleAppsScript.Drive.Folder}
    * @private
    */
    function resolveFolderById_(folderId) {
      if (!folderId || String(folderId).trim() === '') {
        throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'folderId is required.');
      }
      try {
        return DriveApp.getFolderById(folderId);
      } catch (err) {
        throw BIP_UTILS.createError(
          ERROR_CODES.NOT_FOUND,
          `Drive folder with ID "${folderId}" could not be found or accessed: ${err.message || err}`
        );
      }
    }

    return {

      /**
      * Returns metadata for the configured Drive root folder.
      * @return {{folderId: string, folderUrl: string, folderName: string}}
      */
      getRootFolderInfo() {
        return toFolderInfo_(getRootFolder_());
      },

      /**
      * Gets or creates the per-BIP subfolder under the root folder, named
      * from the university and BIP title. Idempotent: calling this again
      * with the same inputs returns the same folder instead of duplicating it.
      * @param {string} universityName
      * @param {string} bipTitle
      * @return {{folderId: string, folderUrl: string, folderName: string}}
      */
      createBipRecordFolder(universityName, bipTitle) {
        const folderName = buildBipFolderName_(universityName, bipTitle);
        const folder = getOrCreateFolder_(getRootFolder_(), folderName);
        BIP_UTILS.logInfo('DriveService', 'Resolved BIP record folder.', {
          folderId: folder.getId(),
          folderName
        });
        return toFolderInfo_(folder);
      },

      /**
      * Looks up an existing per-BIP subfolder without creating one.
      * @param {string} universityName
      * @param {string} bipTitle
      * @return {?{folderId: string, folderUrl: string, folderName: string}}
      */
      findBipRecordFolder(universityName, bipTitle) {
        const folderName = buildBipFolderName_(universityName, bipTitle);
        const folder = findFolderByName_(getRootFolder_(), folderName);
        return folder ? toFolderInfo_(folder) : null;
      },

      /**
      * Saves text/HTML content (e.g. the original source email body) as a
      * new file inside the given folder.
      * @param {string} folderId
      * @param {string} fileName
      * @param {string} content
      * @param {string=} mimeType Defaults to 'text/html'.
      * @return {{fileId: string, fileUrl: string, fileName: string}}
      */
      saveTextFileToFolder(folderId, fileName, content, mimeType) {
        if (!fileName || String(fileName).trim() === '') {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'fileName is required.');
        }
        if (content === null || content === undefined) {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'content is required.');
        }
        const folder = resolveFolderById_(folderId);
        const resolvedMimeType = mimeType || 'text/html';
        const sanitizedFileName = sanitizeName_(fileName, 'Untitled File');
        try {
          const file = folder.createFile(sanitizedFileName, String(content), resolvedMimeType);
          return {
            fileId: file.getId(),
            fileUrl: file.getUrl(),
            fileName: file.getName()
          };
        } catch (err) {
          throw BIP_UTILS.createError(
            ERROR_CODES.DRIVE_ERROR,
            `Unable to save file "${sanitizedFileName}" to folder "${folderId}": ${err.message || err}`
          );
        }
      },

      /**
      * Saves a binary Blob (e.g. a Gmail attachment) as a new file inside
      * the given folder.
      * @param {string} folderId
      * @param {GoogleAppsScript.Base.Blob} blob
      * @return {{fileId: string, fileUrl: string, fileName: string}}
      */
      saveBlobToFolder(folderId, blob) {
        if (!blob) {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'blob is required.');
        }
        const folder = resolveFolderById_(folderId);
        try {
          const file = folder.createFile(blob);
          return {
            fileId: file.getId(),
            fileUrl: file.getUrl(),
            fileName: file.getName()
          };
        } catch (err) {
          throw BIP_UTILS.createError(
            ERROR_CODES.DRIVE_ERROR,
            `Unable to save blob to folder "${folderId}": ${err.message || err}`
          );
        }
      },

      /**
      * Returns the webViewLink-style URL for a folder by its ID.
      * @param {string} folderId
      * @return {string}
      */
      getFolderUrl(folderId) {
        return resolveFolderById_(folderId).getUrl();
      },

      /**
      * Checks whether a folder ID currently resolves to an accessible Drive folder.
      * @param {string} folderId
      * @return {boolean}
      */
      folderExists(folderId) {
        if (!folderId || String(folderId).trim() === '') {
          return false;
        }
        try {
          DriveApp.getFolderById(folderId).getId();
          return true;
        } catch (err) {
          return false;
        }
      },

      /**
      * Lists the direct file children of a folder.
      * @param {string} folderId
      * @return {Array<{fileId: string, fileName: string, fileUrl: string, mimeType: string, createdAt: string}>}
      */
      listFilesInFolder(folderId) {
        const folder = resolveFolderById_(folderId);
        const files = [];
        const iterator = folder.getFiles();
        while (iterator.hasNext()) {
          const file = iterator.next();
          files.push({
            fileId: file.getId(),
            fileName: file.getName(),
            fileUrl: file.getUrl(),
            mimeType: file.getMimeType(),
            createdAt: file.getDateCreated().toISOString()
          });
        }
        return files;
      },

      /**
      * Moves a folder to the Drive trash (soft delete).
      * @param {string} folderId
      * @return {void}
      */
      moveFolderToTrash(folderId) {
        const folder = resolveFolderById_(folderId);
        try {
          folder.setTrashed(true);
          BIP_UTILS.logInfo('DriveService', 'Folder moved to trash.', { folderId });
        } catch (err) {
          throw BIP_UTILS.createError(
            ERROR_CODES.DRIVE_ERROR,
            `Unable to trash folder "${folderId}": ${err.message || err}`
          );
        }
      }
    };

  })());

GmailService.gs

  /**
  * @fileoverview Gmail ingestion layer for the BIP Automation system.
  *
  * GmailService.gs is the SOLE file permitted to interact directly with
  * GmailApp. Every other file that needs to discover or read BIP
  * invitation emails must go exclusively through the `GmailService`
  * object exposed here.
  *
  * ARCHITECTURAL POSITION (Layer 3 - External Data Sources)
  * Depends on Constants.gs, Utilities.gs (BIP_UTILS), Config.gs
  * (ConfigService) for search query / label configuration, and
  * SpreadsheetService.gs (read-only) for duplicate-import checking.
  * Never acquires PropertiesService/CacheService/LockService directly,
  * never writes to Sheets, and never calls GeminiService.
  *
  * STRUCTURED EMAIL SHAPE
  * Every email returned by this service is a plain object with the
  * following shape (property names for pipeline metadata reuse
  * BIP_CONSTANTS.BIP_FIELD_KEYS so downstream services never redeclare
  * the field-key strings):
  *
  * @typedef {Object} StructuredBipEmail
  * @property {string} gmailMessageId        BIP_FIELD_KEYS.GMAIL_MESSAGE_ID
  * @property {string} gmailThreadId         BIP_FIELD_KEYS.GMAIL_THREAD_ID
  * @property {string} sourceEmailSubject    BIP_FIELD_KEYS.SOURCE_EMAIL_SUBJECT
  * @property {Date}   sourceEmailReceivedAt BIP_FIELD_KEYS.SOURCE_EMAIL_RECEIVED_AT
  * @property {string} fromAddress
  * @property {string} plainBody
  * @property {string} htmlBody
  * @property {Array<{fileName: string, contentType: string, sizeBytes: number, blob: GoogleAppsScript.Base.Blob}>} attachments
  *
  * DEDUPLICATION STRATEGY
  * Two independent layers prevent duplicate imports:
  *   1. Gmail-level: processed threads are tagged with the configured
  *      label (Config.getGmailLabelName()) and excluded from subsequent
  *      searches via a `-label:` query clause.
  *   2. Sheet-level: every candidate message is additionally checked
  *      against SpreadsheetService.findRecordByGmailMessageId() in case
  *      a thread was labeled but the write to the sheet never
  *      completed, or vice versa.
  *
  * GLOBAL SCOPE NOTE
  * Exposes exactly ONE global symbol, `GmailService`, built via an IIFE,
  * to avoid collisions in the shared Apps Script global namespace.
  */

  const GmailService = Object.freeze((() => {

    const FIELD_KEYS = BIP_CONSTANTS.BIP_FIELD_KEYS;
    const ERROR_CODES = BIP_CONSTANTS.ERROR_CODES;

    /** Threads fetched per GmailApp.search() page. @private */
    const PAGE_SIZE_ = 25;

    /** Safety cap on total search offset, to bound worst-case execution time/quota. @private */
    const MAX_SEARCH_OFFSET_ = 1000;

    /** Default cap on number of new structured emails returned per call. @private */
    const DEFAULT_MAX_RESULTS_ = 50;

    /**
    * Returns the Gmail label used to mark processed threads, creating it
    * if it does not yet exist.
    * @return {GoogleAppsScript.Gmail.GmailLabel}
    * @private
    */
    function ensureProcessedLabelExists_() {
      const labelName = ConfigService.getGmailLabelName();
      let label;
      try {
        label = GmailApp.getUserLabelByName(labelName);
      } catch (err) {
        throw BIP_UTILS.createError(
          ERROR_CODES.GMAIL_ERROR,
          `Unable to look up Gmail label "${labelName}": ${err.message || err}`
        );
      }
      if (!label) {
        try {
          label = GmailApp.createLabel(labelName);
          BIP_UTILS.logInfo('GmailService', 'Created Gmail label.', { labelName });
        } catch (err) {
          throw BIP_UTILS.createError(
            ERROR_CODES.GMAIL_ERROR,
            `Unable to create Gmail label "${labelName}": ${err.message || err}`
          );
        }
      }
      return label;
    }

    /**
    * Extracts non-inline attachments from a message into plain metadata
    * objects, each carrying the underlying Blob for callers that need to
    * persist it (e.g. via DriveService.saveBlobToFolder).
    * @param {GoogleAppsScript.Gmail.GmailMessage} message
    * @return {Array<{fileName: string, contentType: string, sizeBytes: number, blob: GoogleAppsScript.Base.Blob}>}
    * @private
    */
    function getMessageAttachments_(message) {
      const rawAttachments = message.getAttachments({ includeInlineImages: false, includeAttachments: true });
      return rawAttachments.map((blob) => ({
        fileName: blob.getName(),
        contentType: blob.getContentType(),
        sizeBytes: blob.getBytes().length,
        blob
      }));
    }

    /**
    * Converts a raw GmailMessage into a StructuredBipEmail.
    * @param {GoogleAppsScript.Gmail.GmailMessage} message
    * @return {StructuredBipEmail}
    * @private
    */
    function toStructuredEmail_(message) {
      let attachments;
      try {
        attachments = getMessageAttachments_(message);
      } catch (err) {
        throw BIP_UTILS.createError(
          ERROR_CODES.GMAIL_ERROR,
          `Unable to read attachments for message "${message.getId()}": ${err.message || err}`
        );
      }

      const structured = {};
      structured[FIELD_KEYS.GMAIL_MESSAGE_ID] = message.getId();
      structured[FIELD_KEYS.GMAIL_THREAD_ID] = message.getThread().getId();
      structured[FIELD_KEYS.SOURCE_EMAIL_SUBJECT] = message.getSubject();
      structured[FIELD_KEYS.SOURCE_EMAIL_RECEIVED_AT] = message.getDate();
      structured.fromAddress = message.getFrom();
      structured.plainBody = message.getPlainBody();
      structured.htmlBody = message.getBody();
      structured.attachments = attachments;
      return structured;
    }

    /**
    * Checks whether a Gmail message has already been imported into the
    * "Main" sheet. Duplicate-check failures are treated as non-fatal
    * (message is assumed new) so a transient Sheets error never blocks
    * ingestion entirely; the sheet-level append itself still guards
    * against true duplicates.
    * @param {string} gmailMessageId
    * @return {boolean}
    * @private
    */
    function isAlreadyImported_(gmailMessageId) {
      try {
        return SpreadsheetService.findRecordByGmailMessageId(gmailMessageId) !== null;
      } catch (err) {
        BIP_UTILS.logWarn('GmailService', 'Duplicate check against Main sheet failed; treating message as new.', {
          gmailMessageId,
          error: String(err)
        });
        return false;
      }
    }

    /**
    * Resolves a Gmail message by ID, throwing a well-typed error if it
    * cannot be found or accessed.
    * @param {string} gmailMessageId
    * @return {GoogleAppsScript.Gmail.GmailMessage}
    * @private
    */
    function resolveMessageById_(gmailMessageId) {
      let message;
      try {
        message = GmailApp.getMessageById(gmailMessageId);
      } catch (err) {
        throw BIP_UTILS.createError(
          ERROR_CODES.GMAIL_ERROR,
          `Unable to fetch Gmail message "${gmailMessageId}": ${err.message || err}`
        );
      }
      if (!message) {
        throw BIP_UTILS.createError(
          ERROR_CODES.NOT_FOUND,
          `Gmail message "${gmailMessageId}" was not found.`
        );
      }
      return message;
    }

    return {

      /**
      * Searches Gmail for BIP invitation emails matching the configured
      * search query, excluding threads already tagged as processed, and
      * further excluding any individual message already present in the
      * "Main" sheet. Paginates internally over Gmail search results.
      * @param {number=} maxResults Maximum number of new emails to return (default 50).
      * @return {StructuredBipEmail[]}
      */
      searchNewBipEmails(maxResults) {
        const limit = (Number.isInteger(maxResults) && maxResults > 0) ? maxResults : DEFAULT_MAX_RESULTS_;

        const baseQuery = ConfigService.getGmailSearchQuery();
        if (!baseQuery || String(baseQuery).trim() === '') {
          throw BIP_UTILS.createError(ERROR_CODES.CONFIGURATION_ERROR, 'Gmail search query is not configured.');
        }

        const label = ensureProcessedLabelExists_();
        const searchQuery = `${baseQuery} -label:${label.getName()}`;

        const structuredEmails = [];
        let start = 0;
        let threadsFetchedThisPage = PAGE_SIZE_;

        while (threadsFetchedThisPage === PAGE_SIZE_ && structuredEmails.length < limit && start < MAX_SEARCH_OFFSET_) {
          let threads;
          try {
            threads = GmailApp.search(searchQuery, start, PAGE_SIZE_);
          } catch (err) {
            throw BIP_UTILS.createError(ERROR_CODES.GMAIL_ERROR, `Gmail search failed: ${err.message || err}`);
          }
          threadsFetchedThisPage = threads.length;

          for (let t = 0; t < threads.length && structuredEmails.length < limit; t++) {
            const thread = threads[t];
            let messages;
            try {
              messages = thread.getMessages();
            } catch (err) {
              BIP_UTILS.logWarn('GmailService', 'Failed to read messages for thread; skipping thread.', {
                threadId: thread.getId(),
                error: String(err)
              });
              continue;
            }

            for (let m = 0; m < messages.length && structuredEmails.length < limit; m++) {
              const message = messages[m];
              let messageId;
              try {
                messageId = message.getId();
              } catch (err) {
                BIP_UTILS.logWarn('GmailService', 'Failed to read message ID; skipping message.', { error: String(err) });
                continue;
              }

              if (isAlreadyImported_(messageId)) {
                continue;
              }

              try {
                structuredEmails.push(toStructuredEmail_(message));
              } catch (err) {
                BIP_UTILS.logWarn('GmailService', 'Failed to build structured email; skipping message.', {
                  messageId,
                  error: String(err)
                });
              }
            }
          }

          start += PAGE_SIZE_;
        }

        BIP_UTILS.logInfo('GmailService', 'Completed Gmail search for new BIP emails.', {
          found: structuredEmails.length,
          query: searchQuery
        });

        return structuredEmails;
      },

      /**
      * Fetches a single Gmail message by ID and returns it as a
      * StructuredBipEmail, regardless of its processed/duplicate status.
      * Intended for manual re-parsing (e.g. the "parse" Router action).
      * @param {string} gmailMessageId
      * @return {StructuredBipEmail}
      */
      getEmailByMessageId(gmailMessageId) {
        if (!gmailMessageId || String(gmailMessageId).trim() === '') {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'gmailMessageId is required.');
        }
        const message = resolveMessageById_(gmailMessageId);
        return toStructuredEmail_(message);
      },

      /**
      * Marks a Gmail message's thread as processed by applying the
      * configured label, so it is excluded from future searchNewBipEmails()
      * calls. Call this only after the message has been durably persisted
      * (e.g. appended to the "Main" sheet).
      * @param {string} gmailMessageId
      * @return {void}
      */
      markEmailAsProcessed(gmailMessageId) {
        if (!gmailMessageId || String(gmailMessageId).trim() === '') {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'gmailMessageId is required.');
        }
        const label = ensureProcessedLabelExists_();
        const message = resolveMessageById_(gmailMessageId);
        try {
          message.getThread().addLabel(label);
        } catch (err) {
          throw BIP_UTILS.createError(
            ERROR_CODES.GMAIL_ERROR,
            `Unable to label thread for message "${gmailMessageId}": ${err.message || err}`
          );
        }
        BIP_UTILS.logInfo('GmailService', 'Marked Gmail message as processed.', {
          gmailMessageId,
          label: label.getName()
        });
      },

      /**
      * Marks an entire Gmail thread as processed by applying the
      * configured label. Useful when a thread was found to be a
      * duplicate or invalid and should not be re-surfaced by future
      * searches even though no sheet row was created for it.
      * @param {string} gmailThreadId
      * @return {void}
      */
      markThreadAsProcessed(gmailThreadId) {
        if (!gmailThreadId || String(gmailThreadId).trim() === '') {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'gmailThreadId is required.');
        }
        const label = ensureProcessedLabelExists_();
        let thread;
        try {
          thread = GmailApp.getThreadById(gmailThreadId);
        } catch (err) {
          throw BIP_UTILS.createError(
            ERROR_CODES.GMAIL_ERROR,
            `Unable to fetch Gmail thread "${gmailThreadId}": ${err.message || err}`
          );
        }
        if (!thread) {
          throw BIP_UTILS.createError(ERROR_CODES.NOT_FOUND, `Gmail thread "${gmailThreadId}" was not found.`);
        }
        try {
          thread.addLabel(label);
        } catch (err) {
          throw BIP_UTILS.createError(
            ERROR_CODES.GMAIL_ERROR,
            `Unable to label thread "${gmailThreadId}": ${err.message || err}`
          );
        }
        BIP_UTILS.logInfo('GmailService', 'Marked Gmail thread as processed.', {
          gmailThreadId,
          label: label.getName()
        });
      }
    };

  })());

GeminiService.gs

  /**
  * @fileoverview Gemini Generative Language API client for the BIP
  * Automation system.
  *
  * GeminiService.gs is the SOLE file permitted to call the Gemini API.
  * It converts a raw email body into a validated, structured JSON object
  * keyed by BIP_CONSTANTS.GEMINI_EXTRACTED_FIELD_KEYS. It never touches
  * SpreadsheetApp, DriveApp, or GmailApp, and never depends on
  * GmailService - it operates purely on the plain email content it is
  * given.
  *
  * ARCHITECTURAL POSITION (Layer 3 - External Data Sources)
  * Depends on Constants.gs, Utilities.gs (BIP_UTILS), and Config.gs
  * (ConfigService) for the API key, model name, and request defaults.
  * Never acquires PropertiesService/CacheService/LockService directly.
  *
  * RELIABILITY
  * Transient failures (HTTP 429, HTTP 5xx, network exceptions) are
  * retried up to BIP_CONSTANTS.GEMINI_DEFAULTS.MAX_RETRIES times with
  * exponential backoff seeded from RETRY_BASE_DELAY_MS. Non-transient
  * failures (4xx other than 429, malformed responses, schema
  * violations) fail immediately with a typed GEMINI_ERROR.
  *
  * GLOBAL SCOPE NOTE
  * Exposes exactly ONE global symbol, `GeminiService`, built via an IIFE,
  * to avoid collisions in the shared Apps Script global namespace.
  */

  const GeminiService = Object.freeze((() => {

    const FIELD_KEYS = BIP_CONSTANTS.BIP_FIELD_KEYS;
    const EXTRACTED_KEYS = BIP_CONSTANTS.GEMINI_EXTRACTED_FIELD_KEYS;
    const ACADEMIC_LEVELS = BIP_CONSTANTS.ACADEMIC_LEVELS;

    const ERROR_CODES = BIP_CONSTANTS.ERROR_CODES;

    /** Maximum characters of any external text kept in error/log messages. @private */
    const MAX_LOGGED_TEXT_LENGTH_ = 500;

    /**
    * Human-readable descriptions of each extracted field, used to build
    * both the extraction prompt and the response schema. Keyed by
    * BIP_CONSTANTS.BIP_FIELD_KEYS values so it can never drift from the
    * canonical field-key list.
    * @private
    */
    const FIELD_DESCRIPTIONS_ = Object.freeze({
      [FIELD_KEYS.BIP_TITLE]: 'The official title or name of the Blended Intensive Programme (BIP).',
      [FIELD_KEYS.UNIVERSITY]: 'The name of the university or institution hosting/organizing the programme.',
      [FIELD_KEYS.COUNTRY]: 'The country where the physical mobility of the programme takes place.',
      [FIELD_KEYS.ACADEMIC_LEVEL]: `The academic level of eligible participants. Must be exactly one of: ${VALID_ACADEMIC_LEVELS.join(', ')}.`,
      [FIELD_KEYS.START_DATE]: 'The start date of the physical/blended mobility period, formatted strictly as yyyy-MM-dd.',
      [FIELD_KEYS.END_DATE]: 'The end date of the physical/blended mobility period, formatted strictly as yyyy-MM-dd.',
      [FIELD_KEYS.APPLICATION_DEADLINE]: 'The deadline by which students must submit their application, formatted strictly as yyyy-MM-dd.',
      [FIELD_KEYS.APPLICATION_METHOD]: 'How students should apply: a URL, an email address, or a short description of the process.',
      [FIELD_KEYS.SUITABLE_FACULTIES]: 'An array of faculty names or study fields this programme is suitable for.',
      [FIELD_KEYS.ADDITIONAL_NOTES]: 'Any other relevant details not captured by the other fields (funding, ECTS credits, language of instruction, etc.).'
    });

    /**
    * Truncates text for safe inclusion in log messages and thrown errors.
    * @param {string} text
    * @param {number=} maxLength
    * @return {string}
    * @private
    */
    function safeTruncate_(text, maxLength) {
      const limit = maxLength || MAX_LOGGED_TEXT_LENGTH_;
      const str = String(text === undefined || text === null ? '' : text);
      return str.length > limit ? `${str.substring(0, limit)}...[truncated]` : str;
    }

    /**
    * Builds the Gemini responseSchema object describing the exact shape
    * of the extraction result, derived from BIP_CONSTANTS.GEMINI_EXTRACTED_FIELD_KEYS
    * so the schema can never omit or duplicate a field.
    * @return {Object}
    * @private
    */
    function buildResponseSchema_() {
      const properties = {};

      EXTRACTED_KEYS.forEach((fieldKey) => {
        if (fieldKey === FIELD_KEYS.SUITABLE_FACULTIES) {
          properties[fieldKey] = {
            type: 'ARRAY',
            items: { type: 'STRING' },
            description: FIELD_DESCRIPTIONS_[fieldKey]
          };
        } else if (fieldKey === FIELD_KEYS.ACADEMIC_LEVEL) {
          properties[fieldKey] = {
            type: 'STRING',
            enum: VALID_ACADEMIC_LEVELS,
            description: FIELD_DESCRIPTIONS_[fieldKey]
          };
        } else {
          properties[fieldKey] = {
            type: 'STRING',
            description: FIELD_DESCRIPTIONS_[fieldKey]
          };
        }
      });

      return {
        type: 'OBJECT',
        properties,
        required: EXTRACTED_KEYS.slice()
      };
    }

    /**
    * Builds the natural-language extraction prompt for a single email.
    * @param {{subject: ?string, fromAddress: ?string, body: string}} emailContext
    * @return {string}
    * @private
    */
    function buildExtractionPrompt_(emailContext) {
      const fieldList = EXTRACTED_KEYS
        .map((fieldKey) => `- "${fieldKey}": ${FIELD_DESCRIPTIONS_[fieldKey]}`)
        .join('\n');

      const subject = emailContext.subject || '(no subject)';
      const fromAddress = emailContext.fromAddress || '(unknown sender)';
      const body = emailContext.body || '';

      return [
        'You are an information-extraction assistant for a university International Relations Office.',
        'You will be given the content of one email that announces or invites students to an Erasmus+ Blended Intensive Programme (BIP).',
        'Extract the following fields and return ONLY a JSON object with exactly these keys - no markdown, no explanation, no code fences:',
        fieldList,
        'Rules:',
        '- If a date cannot be determined with confidence, use an empty string "" rather than guessing.',
        `- If the academic level cannot be determined, use "${ACADEMIC_LEVELS.NOT_SPECIFIED}".`,
        '- "suitableFaculties" must always be a JSON array of strings, even if empty ([]) or containing a single item.',
        '- Dates must be formatted strictly as yyyy-MM-dd.',
        '- Do not invent information that is not present in the email.',
        '',
        `Email subject: ${subject}`,
        `Email sender: ${fromAddress}`,
        'Email content:',
        '"""',
        body,
        '"""'
      ].join('\n');
    }

    /**
    * Builds the full Gemini generateContent request payload for a given prompt.
    * @param {string} promptText
    * @return {Object}
    * @private
    */
    function buildExtractionRequestPayload_(promptText) {
      return {
        contents: [
          {
            role: 'user',
            parts: [{ text: promptText }]
          }
        ],
        generationConfig: {
          temperature: GEMINI_DEFAULTS.TEMPERATURE,
          maxOutputTokens: GEMINI_DEFAULTS.MAX_OUTPUT_TOKENS,
          responseMimeType: GEMINI_DEFAULTS.RESPONSE_MIME_TYPE,
          responseSchema: buildResponseSchema_()
        }
      };
    }

    /**
    * Builds the fully-qualified Gemini generateContent endpoint URL,
    * reading the model and API key from ConfigService.
    * @return {string}
    * @private
    */
    function buildEndpointUrl_() {
      const model = ConfigService.getGeminiModel();
      const apiKey = ConfigService.getGeminiApiKey();
      return `${GEMINI_DEFAULTS.API_BASE_URL}/${GEMINI_DEFAULTS.API_VERSION}/models/${model}:${GEMINI_DEFAULTS.GENERATE_CONTENT_METHOD}?key=${encodeURIComponent(apiKey)}`;
    }

    /**
    * Determines whether an HTTP status code represents a transient
    * failure worth retrying.
    * @param {number} statusCode
    * @return {boolean}
    * @private
    */
    function isTransientStatusCode_(statusCode) {
      return statusCode === 429 || (statusCode >= 500 && statusCode < 600);
    }

    /**
    * Sleeps with exponential backoff ahead of a retry attempt.
    * @param {number} attempt 1-based attempt number that just failed.
    * @private
    */
    function sleepBeforeRetry_(attempt) {
      const delayMs = GEMINI_DEFAULTS.RETRY_BASE_DELAY_MS * Math.pow(2, attempt - 1);
      Utilities.sleep(delayMs);
    }

    /**
    * Performs the Gemini API call with retry-on-transient-failure logic.
    * @param {Object} payload Request body to POST as JSON.
    * @return {GoogleAppsScript.URL_Fetch.HTTPResponse} The successful (HTTP 200) response.
    * @private
    */
    function fetchWithRetry_(payload) {
      const url = buildEndpointUrl_();
      const options = {
        method: 'post',
        contentType: 'application/json',
        payload: JSON.stringify(payload),
        muteHttpExceptions: true
      };

      let lastNetworkError = null;

      for (let attempt = 1; attempt <= GEMINI_DEFAULTS.MAX_RETRIES; attempt++) {
        let response;
        try {
          response = UrlFetchApp.fetch(url, options);
        } catch (networkErr) {
          lastNetworkError = networkErr;
          BIP_UTILS.logWarn('GeminiService', 'Network error calling Gemini API; will retry if attempts remain.', {
            attempt,
            error: String(networkErr)
          });
          if (attempt < GEMINI_DEFAULTS.MAX_RETRIES) {
            sleepBeforeRetry_(attempt);
          }
          continue;
        }

        const statusCode = response.getResponseCode();
        if (statusCode === 200) {
          return response;
        }

        if (isTransientStatusCode_(statusCode) && attempt < GEMINI_DEFAULTS.MAX_RETRIES) {
          BIP_UTILS.logWarn('GeminiService', 'Transient Gemini API error; retrying.', {
            attempt,
            statusCode,
            body: safeTruncate_(response.getContentText())
          });
          sleepBeforeRetry_(attempt);
          continue;
        }

        throw BIP_UTILS.createError(
          ERROR_CODES.GEMINI_ERROR,
          `Gemini API returned HTTP ${statusCode}: ${safeTruncate_(response.getContentText())}`
        );
      }

      throw BIP_UTILS.createError(
        ERROR_CODES.GEMINI_ERROR,
        `Gemini API call failed after ${GEMINI_DEFAULTS.MAX_RETRIES} attempt(s): ${lastNetworkError ? (lastNetworkError.message || lastNetworkError) : 'unknown error'}`
      );
    }

    /**
    * Strips defensive markdown code-fence wrapping from model output, in
    * case the model wraps its JSON in ``` fences despite responseMimeType.
    * @param {string} text
    * @return {string}
    * @private
    */
    function stripMarkdownFences_(text) {
      let result = String(text).trim();
      if (result.startsWith('```')) {
        result = result.replace(/^```(json)?/i, '').replace(/```$/, '').trim();
      }
      return result;
    }

    /**
    * Parses and validates the raw Gemini API envelope, returning the
    * inner extraction JSON (not yet field-normalized).
    * @param {GoogleAppsScript.URL_Fetch.HTTPResponse} response
    * @return {*} The parsed inner JSON value (expected to be an object).
    * @private
    */
    function parseApiResponse_(response) {
      let envelope;
      try {
        envelope = JSON.parse(response.getContentText());
      } catch (err) {
        throw BIP_UTILS.createError(
          ERROR_CODES.GEMINI_ERROR,
          `Gemini API returned invalid JSON envelope: ${err.message || err}`
        );
      }

      if (envelope.promptFeedback && envelope.promptFeedback.blockReason) {
        throw BIP_UTILS.createError(
          ERROR_CODES.GEMINI_ERROR,
          `Gemini blocked the request: ${envelope.promptFeedback.blockReason}`
        );
      }

      if (!envelope.candidates || !Array.isArray(envelope.candidates) || envelope.candidates.length === 0) {
        throw BIP_UTILS.createError(ERROR_CODES.GEMINI_ERROR, 'Gemini API response contained no candidates.');
      }

      const candidate = envelope.candidates[0];
      if (candidate.finishReason && candidate.finishReason !== 'STOP' && candidate.finishReason !== 'MAX_TOKENS') {
        throw BIP_UTILS.createError(
          ERROR_CODES.GEMINI_ERROR,
          `Gemini generation did not complete normally: ${candidate.finishReason}`
        );
      }

      const parts = candidate.content && candidate.content.parts;
      if (!parts || !Array.isArray(parts) || parts.length === 0) {
        throw BIP_UTILS.createError(ERROR_CODES.GEMINI_ERROR, 'Gemini API response did not contain any content parts.');
      }

      const rawText = parts
        .map((part) => (typeof part.text === 'string' ? part.text : ''))
        .join('')
        .trim();

      if (rawText === '') {
        throw BIP_UTILS.createError(ERROR_CODES.GEMINI_ERROR, 'Gemini API response text was empty.');
      }

      const cleanedText = stripMarkdownFences_(rawText);

      try {
        return JSON.parse(cleanedText);
      } catch (err) {
        throw BIP_UTILS.createError(
          ERROR_CODES.GEMINI_ERROR,
          `Gemini did not return valid JSON: ${err.message || err}. Raw text: ${safeTruncate_(cleanedText)}`
        );
      }
    }

    /**
    * Normalizes and validates a raw parsed extraction object into a clean
    * object containing exactly BIP_CONSTANTS.GEMINI_EXTRACTED_FIELD_KEYS,
    * with safe defaults for any missing or malformed field.
    * @param {*} rawParsed
    * @return {Object<string, (string|string[])>}
    * @private
    */
    function normalizeExtractedFields_(rawParsed) {
      if (!rawParsed || typeof rawParsed !== 'object' || Array.isArray(rawParsed)) {
        throw BIP_UTILS.createError(ERROR_CODES.GEMINI_ERROR, 'Gemini extraction result must be a JSON object.');
      }

      const normalized = {};

      EXTRACTED_KEYS.forEach((fieldKey) => {
        if (fieldKey === FIELD_KEYS.SUITABLE_FACULTIES) {
          const value = rawParsed[fieldKey];
          if (Array.isArray(value)) {
            normalized[fieldKey] = value
              .map((item) => String(item).trim())
              .filter((item) => item !== '');
          } else if (typeof value === 'string' && value.trim() !== '') {
            normalized[fieldKey] = [value.trim()];
          } else {
            normalized[fieldKey] = [];
          }
          return;
        }

        const rawValue = rawParsed[fieldKey];
        if (rawValue === undefined || rawValue === null) {
          BIP_UTILS.logWarn('GeminiService', `Gemini omitted field "${fieldKey}"; defaulting to empty value.`, {});
          normalized[fieldKey] = (fieldKey === FIELD_KEYS.ACADEMIC_LEVEL) ? ACADEMIC_LEVELS.NOT_SPECIFIED : '';
          return;
        }

        normalized[fieldKey] = String(rawValue).trim();
      });

      if (VALID_ACADEMIC_LEVELS.indexOf(normalized[FIELD_KEYS.ACADEMIC_LEVEL]) === -1) {
        BIP_UTILS.logWarn('GeminiService', 'Gemini returned an unrecognized academic level; defaulting to "Not specified".', {
          received: normalized[FIELD_KEYS.ACADEMIC_LEVEL]
        });
        normalized[FIELD_KEYS.ACADEMIC_LEVEL] = ACADEMIC_LEVELS.NOT_SPECIFIED;
      }

      [FIELD_KEYS.START_DATE, FIELD_KEYS.END_DATE, FIELD_KEYS.APPLICATION_DEADLINE].forEach((dateField) => {
        const value = normalized[dateField];
        if (value !== '' && !BIP_UTILS.isValidIsoDate(value)) {
          BIP_UTILS.logWarn('GeminiService', `Gemini returned a non-ISO date for "${dateField}"; clearing it.`, {
            received: value
          });
          normalized[dateField] = '';
        }
      });

      return normalized;
    }

    return {

      /**
      * Extracts structured BIP fields from a single email's content using
      * the Gemini API. Returns exactly the keys in
      * BIP_CONSTANTS.GEMINI_EXTRACTED_FIELD_KEYS, with safe defaults
      * substituted for any field Gemini could not determine.
      * @param {{subject: ?string, fromAddress: ?string, body: ?string, plainBody: ?string}} emailContext
      *   `body` (or `plainBody` as a fallback) must contain the email text to analyze.
      * @return {Object<string, (string|string[])>} Normalized extraction result.
      */
      extractBipFieldsFromEmail(emailContext) {
        if (!emailContext || typeof emailContext !== 'object') {
          throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'emailContext is required.');
        }

        const body = emailContext.body || emailContext.plainBody;
        if (!body || String(body).trim() === '') {
          throw BIP_UTILS.createError(
            ERROR_CODES.VALIDATION_ERROR,
            'emailContext.body (or plainBody) is required and cannot be empty.'
          );
        }

        const promptText = buildExtractionPrompt_({
          subject: emailContext.subject,
          fromAddress: emailContext.fromAddress,
          body
        });
        const payload = buildExtractionRequestPayload_(promptText);

        BIP_UTILS.logInfo('GeminiService', 'Sending extraction request to Gemini.', {
          model: ConfigService.getGeminiModel(),
          subject: emailContext.subject || null
        });

        const response = fetchWithRetry_(payload);
        const rawParsed = parseApiResponse_(response);
        const normalized = normalizeExtractedFields_(rawParsed);

        BIP_UTILS.logInfo('GeminiService', 'Successfully extracted BIP fields from email.', {
          bipTitle: normalized[FIELD_KEYS.BIP_TITLE] || null
        });

        return normalized;
      },

      /**
      * Performs a minimal round-trip call to the Gemini API to verify
      * that the configured API key and model are valid and reachable.
      * Intended for the ROUTER_ACTIONS.TEST_GEMINI Router action.
      * @return {{success: boolean, model: string, latencyMs: number}}
      */
      testConnection() {
        const model = ConfigService.getGeminiModel();
        const startedAt = Date.now();

        const payload = {
          contents: [
            {
              role: 'user',
              parts: [{ text: 'Reply with only this exact JSON object and nothing else: {"status":"ok"}' }]
            }
          ],
          generationConfig: {
            temperature: 0,
            maxOutputTokens: 32,
            responseMimeType: GEMINI_DEFAULTS.RESPONSE_MIME_TYPE
          }
        };

        const response = fetchWithRetry_(payload);
        const parsed = parseApiResponse_(response);
        const latencyMs = Date.now() - startedAt;

        if (!parsed || parsed.status !== 'ok') {
          throw BIP_UTILS.createError(
            ERROR_CODES.GEMINI_ERROR,
            `Gemini connectivity test returned an unexpected response: ${safeTruncate_(JSON.stringify(parsed))}`
          );
        }

        BIP_UTILS.logInfo('GeminiService', 'Gemini connectivity test succeeded.', { model, latencyMs });

        return { success: true, model, latencyMs };
      }
    };

  })());

ParserService.gs

  /**
  * @fileoverview Extraction-to-record transformation layer for the BIP
  * Automation system.
  *
  * ParserService.gs converts the raw, free-text-ish JSON produced by
  * GeminiService into a fully validated, normalized BIP record DTO ready
  * to be persisted by SpreadsheetService. It is a pure data-transformation
  * layer: given the same inputs it always produces the same output (or
  * throws), and it performs no I/O of any kind.
  *
  * ARCHITECTURAL POSITION (Layer 3 - Data Transformation)
  * Depends only on Constants.gs and Utilities.gs (BIP_UTILS), the same as
  * GmailService.gs and GeminiService.gs. It never calls GmailApp,
  * SpreadsheetApp, DriveApp, UrlFetchApp, or any other Google Apps Script
  * service, and it never calls another BIP_* service. Callers (currently
  * AutomationService.gs) are responsible for supplying every input this
  * file needs, including the list of known STU faculty names used for
  * faculty resolution - ParserService never reads the "Faculty
  * Coordinators" sheet itself.
  *
  * NORMALIZATION SCOPE NOTE
  * Gemini AI already applies light normalization (date format, academic
  * level enum, array coercion for suitable faculties) before this file
  * ever sees the data - see GeminiService.gs's normalizeExtractedFields_.
  * ParserService re-validates and further normalizes defensively so that
  * (a) a future caller that bypasses GeminiService's own normalization
  * (e.g. a manually re-submitted extraction) is still handled safely, and
  * (b) domain-specific normalization that does not belong in a generic
  * AI-extraction client (STU faculty resolution, country alias
  * canonicalization, university name cleanup) lives in exactly one place.
  *
  * REQUIRED-FIELD POLICY
  * BIP_TITLE, UNIVERSITY, COUNTRY, APPLICATION_METHOD and
  * APPLICATION_DEADLINE are treated as mandatory: without them a record
  * is not actionable by a faculty coordinator, and APPLICATION_DEADLINE
  * specifically is what drives the entire reminder workflow. START_DATE
  * and END_DATE may legitimately be unknown at invitation time (Gemini is
  * explicitly instructed to return "" rather than guess), so they are
  * validated for ISO format only when present, not required outright.
  *
  * GLOBAL SCOPE NOTE
  * Exposes exactly ONE global symbol, `ParserService`, built via an IIFE,
  * to avoid collisions in the shared Apps Script global namespace.
  */

  const ParserService = Object.freeze((() => {

    const FIELD_KEYS = BIP_CONSTANTS.BIP_FIELD_KEYS;
    const GEMINI_EXTRACTED_KEYS = BIP_CONSTANTS.GEMINI_EXTRACTED_FIELD_KEYS;
    const ACADEMIC_LEVELS = BIP_CONSTANTS.ACADEMIC_LEVELS;
    const VALID_ACADEMIC_LEVELS = BIP_CONSTANTS.VALID_ACADEMIC_LEVELS;
    const ERROR_CODES = BIP_CONSTANTS.ERROR_CODES;

    /** Maximum stored length for free-text fields, to keep sheet cells manageable. @private */
    const MAX_TITLE_LENGTH_ = 300;
    const MAX_APPLICATION_METHOD_LENGTH_ = 500;
    const MAX_ADDITIONAL_NOTES_LENGTH_ = 2000;

    /** Separator used when persisting the resolved faculty list into a single sheet cell. @private */
    const SUITABLE_FACULTIES_JOIN_SEPARATOR_ = ', ';

    /**
    * Alias table mapping common free-text academic-level variants (already
    * lowercased and stripped of non-alphanumeric characters by
    * normalizeAliasKey_) to the canonical BIP_CONSTANTS.ACADEMIC_LEVELS
    * value. Extend this map, never add a second normalization path.
    * @private
    */
    const ACADEMIC_LEVEL_ALIASES_ = Object.freeze({
      bachelor: ACADEMIC_LEVELS.BACHELOR,
      bc: ACADEMIC_LEVELS.BACHELOR,
      bsc: ACADEMIC_LEVELS.BACHELOR,
      ba: ACADEMIC_LEVELS.BACHELOR,
      undergraduate: ACADEMIC_LEVELS.BACHELOR,
      bakalar: ACADEMIC_LEVELS.BACHELOR,
      bakalarsky: ACADEMIC_LEVELS.BACHELOR,
      master: ACADEMIC_LEVELS.MASTER,
      msc: ACADEMIC_LEVELS.MASTER,
      ma: ACADEMIC_LEVELS.MASTER,
      ing: ACADEMIC_LEVELS.MASTER,
      graduate: ACADEMIC_LEVELS.MASTER,
      postgraduate: ACADEMIC_LEVELS.MASTER,
      magister: ACADEMIC_LEVELS.MASTER,
      magistersky: ACADEMIC_LEVELS.MASTER,
      inziniersky: ACADEMIC_LEVELS.MASTER,
      phd: ACADEMIC_LEVELS.PHD,
      dr: ACADEMIC_LEVELS.PHD,
      doctoral: ACADEMIC_LEVELS.PHD,
      doctorate: ACADEMIC_LEVELS.PHD,
      doktorandsky: ACADEMIC_LEVELS.PHD
    });

    /**
    * Alias table mapping common free-text country variants (already
    * lowercased and stripped of non-alphanumeric characters by
    * normalizeAliasKey_) to a canonical English display name. Countries
    * not present here fall back to a plain title-cased rendering of the
    * trimmed input. Extend this map, never add a second normalization path.
    * @private
    */
    const COUNTRY_ALIASES_ = Object.freeze({
      usa: 'United States',
      us: 'United States',
      unitedstatesofamerica: 'United States',
      uk: 'United Kingdom',
      greatbritain: 'United Kingdom',
      england: 'United Kingdom',
      czechia: 'Czech Republic',
      czechrepublic: 'Czech Republic',
      holland: 'Netherlands',
      thenetherlands: 'Netherlands',
      uae: 'United Arab Emirates',
      southkorea: 'South Korea',
      koreasouth: 'South Korea',
      slovakia: 'Slovakia',
      slovakrepublic: 'Slovakia'
    });

    /**
    * Collapses internal whitespace runs to a single space and trims the
    * result.
    * @param {*} text
    * @return {string}
    * @private
    */
    function collapseWhitespace_(text) {
      return String(text === undefined || text === null ? '' : text).replace(/\s+/g, ' ').trim();
    }

    /**
    * Produces a diacritic-free, lowercase, alphanumeric-only key suitable
    * for exact alias lookups (e.g. "Bc." -> "bc", "Czechia" -> "czechia").
    * @param {*} text
    * @return {string}
    * @private
    */
    function normalizeAliasKey_(text) {
      return String(text === undefined || text === null ? '' : text)
        .normalize('NFD')
        .replace(/[\u0300-\u036f]/g, '')
        .toLowerCase()
        .replace(/[^a-z0-9]/g, '');
    }

    /**
    * Produces a diacritic-free, lowercase key with word boundaries
    * preserved as single spaces, suitable for substring/token matching
    * (e.g. faculty name resolution).
    * @param {*} text
    * @return {string}
    * @private
    */
    function normalizeForMatching_(text) {
      return String(text === undefined || text === null ? '' : text)
        .normalize('NFD')
        .replace(/[\u0300-\u036f]/g, '')
        .toLowerCase()
        .replace(/[^a-z0-9]+/g, ' ')
        .trim();
    }

    /**
    * Title-cases a plain string ("united states" -> "United States").
    * @param {string} text
    * @return {string}
    * @private
    */
    function toTitleCase_(text) {
      return text
        .toLowerCase()
        .split(' ')
        .map((word) => (word.length > 0 ? word.charAt(0).toUpperCase() + word.slice(1) : word))
        .join(' ');
    }

    /**
    * Coerces a Gemini "suitableFaculties" value into an array of strings,
    * defensively accepting a comma-separated string as a fallback shape.
    * @param {*} value
    * @return {string[]}
    * @private
    */
    function coerceToArray_(value) {
      if (Array.isArray(value)) {
        return value;
      }
      if (BIP_UTILS.isNonEmptyString(value)) {
        return value.split(',');
      }
      return [];
    }

    /**
    * Finds the best matching known STU faculty name for a single raw,
    * free-text faculty/field-of-study string. Tries an exact
    * (diacritic-insensitive, case-insensitive) match first, then falls
    * back to a bidirectional substring/token match. Returns null if no
    * known faculty is a plausible match.
    * @param {string} rawEntry Already-trimmed raw faculty text.
    * @param {string[]} knownList Known STU faculty names.
    * @return {?string}
    * @private
    */
    function findBestFacultyMatch_(rawEntry, knownList) {
      const normalizedRaw = normalizeForMatching_(rawEntry);
      if (normalizedRaw === '') {
        return null;
      }

      for (let i = 0; i < knownList.length; i++) {
        if (normalizeForMatching_(knownList[i]) === normalizedRaw) {
          return knownList[i];
        }
      }

      for (let i = 0; i < knownList.length; i++) {
        const normalizedKnown = normalizeForMatching_(knownList[i]);
        if (normalizedKnown !== '' && (normalizedKnown.indexOf(normalizedRaw) !== -1 || normalizedRaw.indexOf(normalizedKnown) !== -1)) {
          return knownList[i];
        }
      }

      return null;
    }

    /**
    * Validates and normalizes a single ISO-date field extracted by Gemini.
    * @param {*} rawValue
    * @param {string} fieldLabel Used in error messages (a BIP_FIELD_KEYS value).
    * @param {boolean} isRequired Whether an empty value should throw.
    * @return {string} Empty string, or a valid yyyy-MM-dd string.
    * @throws {BIP_UTILS.AppError}
    * @private
    */
    function normalizeIsoDate_(rawValue, fieldLabel, isRequired) {
      const trimmed = BIP_UTILS.isNonEmptyString(rawValue) ? rawValue.trim() : '';
      if (trimmed === '') {
        if (isRequired) {
          throw BIP_UTILS.createError(
            ERROR_CODES.VALIDATION_ERROR,
            `Field "${fieldLabel}" is required and could not be determined from the source email.`
          );
        }
        return '';
      }
      if (!BIP_UTILS.isValidIsoDate(trimmed)) {
        throw BIP_UTILS.createError(
          ERROR_CODES.VALIDATION_ERROR,
          `Field "${fieldLabel}" must be a valid ISO date (yyyy-MM-dd). Received: "${trimmed}".`
        );
      }
      return trimmed;
    }

    /**
    * Ensures END_DATE is not chronologically before START_DATE when both
    * are known.
    * @param {string} startDate
    * @param {string} endDate
    * @throws {BIP_UTILS.AppError}
    * @private
    */
    function assertChronologicalOrder_(startDate, endDate) {
      if (startDate === '' || endDate === '') {
        return;
      }
      const start = BIP_UTILS.parseIsoDate(startDate);
      const end = BIP_UTILS.parseIsoDate(endDate);
      if (start && end && end.getTime() < start.getTime()) {
        throw BIP_UTILS.createError(
          ERROR_CODES.VALIDATION_ERROR,
          `Field "${FIELD_KEYS.END_DATE}" ("${endDate}") cannot be earlier than "${FIELD_KEYS.START_DATE}" ("${startDate}").`
        );
      }
    }

    /**
    * Normalizes a BIP title: collapses whitespace, strips wrapping quotes,
    * and enforces a maximum length.
    * @param {*} rawValue
    * @return {string}
    * @private
    */
    function normalizeBipTitle_(rawValue) {
      if (!BIP_UTILS.isNonEmptyString(rawValue)) {
        return '';
      }
      const cleaned = collapseWhitespace_(rawValue).replace(/^["'`]+|["'`]+$/g, '');
      return BIP_UTILS.truncateString(cleaned, MAX_TITLE_LENGTH_);
    }

    /**
    * Normalizes the application method field: collapses whitespace and
    * enforces a maximum length. Case is preserved since the value may be a
    * URL or email address.
    * @param {*} rawValue
    * @return {string}
    * @private
    */
    function normalizeApplicationMethod_(rawValue) {
      if (!BIP_UTILS.isNonEmptyString(rawValue)) {
        return '';
      }
      return BIP_UTILS.truncateString(collapseWhitespace_(rawValue), MAX_APPLICATION_METHOD_LENGTH_);
    }

    /**
    * Normalizes the additional-notes field: collapses whitespace and
    * enforces a maximum length.
    * @param {*} rawValue
    * @return {string}
    * @private
    */
    function normalizeAdditionalNotes_(rawValue) {
      if (!BIP_UTILS.isNonEmptyString(rawValue)) {
        return '';
      }
      return BIP_UTILS.truncateString(collapseWhitespace_(rawValue), MAX_ADDITIONAL_NOTES_LENGTH_);
    }

    /**
    * Verifies that a Gemini extraction result contains every key defined
    * in BIP_CONSTANTS.GEMINI_EXTRACTED_FIELD_KEYS.
    * @param {Object} extractedFields
    * @throws {BIP_UTILS.AppError}
    * @private
    */
    function assertCompleteExtraction_(extractedFields) {
      const missingKeys = GEMINI_EXTRACTED_KEYS.filter(
        (key) => !Object.prototype.hasOwnProperty.call(extractedFields, key)
      );
      if (missingKeys.length > 0) {
        throw BIP_UTILS.createError(
          ERROR_CODES.VALIDATION_ERROR,
          `Gemini extraction result is missing required key(s): ${missingKeys.join(', ')}.`
        );
      }
    }

    /**
    * Normalizes a free-text academic level string into one of
    * BIP_CONSTANTS.ACADEMIC_LEVELS. Falls back to
    * ACADEMIC_LEVELS.NOT_SPECIFIED for empty or unrecognized input.
    * @param {*} rawValue
    * @return {string} One of BIP_CONSTANTS.ACADEMIC_LEVELS.
    */
    function normalizeAcademicLevel(rawValue) {
      if (!BIP_UTILS.isNonEmptyString(rawValue)) {
        return ACADEMIC_LEVELS.NOT_SPECIFIED;
      }

      const trimmed = rawValue.trim();
      for (let i = 0; i < VALID_ACADEMIC_LEVELS.length; i++) {
        if (VALID_ACADEMIC_LEVELS[i].toLowerCase() === trimmed.toLowerCase()) {
          return VALID_ACADEMIC_LEVELS[i];
        }
      }

      const aliasKey = normalizeAliasKey_(trimmed);
      if (Object.prototype.hasOwnProperty.call(ACADEMIC_LEVEL_ALIASES_, aliasKey)) {
        return ACADEMIC_LEVEL_ALIASES_[aliasKey];
      }

      BIP_UTILS.logWarn('ParserService', 'Unrecognized academic level value; defaulting to "Not specified".', {
        received: rawValue
      });
      return ACADEMIC_LEVELS.NOT_SPECIFIED;
    }

    /**
    * Normalizes a free-text country name: resolves known aliases to a
    * canonical display name, otherwise title-cases the trimmed input.
    * @param {*} rawValue
    * @return {string} Empty string if rawValue is blank.
    */
    function normalizeCountry(rawValue) {
      if (!BIP_UTILS.isNonEmptyString(rawValue)) {
        return '';
      }
      const trimmed = collapseWhitespace_(rawValue);
      const aliasKey = normalizeAliasKey_(trimmed);
      if (Object.prototype.hasOwnProperty.call(COUNTRY_ALIASES_, aliasKey)) {
        return COUNTRY_ALIASES_[aliasKey];
      }
      return toTitleCase_(trimmed);
    }

    /**
    * Normalizes a university/institution name: collapses whitespace and
    * strips wrapping quote characters. Casing is preserved to avoid
    * mangling acronyms and proper nouns.
    * @param {*} rawValue
    * @return {string} Empty string if rawValue is blank.
    */
    function normalizeUniversityName(rawValue) {
      if (!BIP_UTILS.isNonEmptyString(rawValue)) {
        return '';
      }
      return collapseWhitespace_(rawValue).replace(/^["'`]+|["'`]+$/g, '');
    }

    /**
    * Resolves a list of free-text faculty/field-of-study strings extracted
    * by Gemini against the list of known STU faculty names, deduplicating
    * the result. Entries that cannot be confidently matched are kept as
    * their trimmed, original text (so information is never silently
    * dropped) and a warning is logged.
    * @param {*} rawFaculties Array (or comma-separated string) of raw faculty text.
    * @param {string[]=} knownFacultyNames Known STU faculty names (e.g. from the "Faculty Coordinators" sheet).
    * @return {string[]} Deduplicated, resolved faculty names.
    */
    function resolveSuitableFaculties(rawFaculties, knownFacultyNames) {
      const knownList = Array.isArray(knownFacultyNames)
        ? knownFacultyNames.filter((name) => BIP_UTILS.isNonEmptyString(name))
        : [];
      const rawList = coerceToArray_(rawFaculties);

      const resolved = [];
      const seenKeys = {};

      rawList.forEach((rawEntry) => {
        if (!BIP_UTILS.isNonEmptyString(rawEntry)) {
          return;
        }
        const trimmedRaw = rawEntry.trim();
        const match = findBestFacultyMatch_(trimmedRaw, knownList);
        const finalName = match || trimmedRaw;
        const dedupeKey = normalizeAliasKey_(finalName);

        if (dedupeKey === '' || seenKeys[dedupeKey]) {
          return;
        }
        seenKeys[dedupeKey] = true;
        resolved.push(finalName);

        if (!match) {
          BIP_UTILS.logWarn('ParserService', 'Suitable faculty could not be matched to a known STU faculty; keeping the extracted value as-is.', {
            rawEntry: trimmedRaw
          });
        }
      });

      return resolved;
    }

    /**
    * @typedef {Object} BipRecordDto
    * A plain object keyed by BIP_CONSTANTS.BIP_FIELD_KEYS values, ready to
    * be persisted via SpreadsheetService.appendBipRecord. Does not include
    * DRIVE_FOLDER_URL, DECISION_STATUS, LAST_REMINDER_SENT_AT or
    * REMINDER_COUNT, since those are either set later in the pipeline
    * (Drive folder URL) or safely defaulted by SpreadsheetService itself.
    */

    /**
    * Builds a fully validated, normalized BipRecordDto from a single
    * Gmail-sourced structured email and its Gemini extraction result.
    * @param {Object} structuredEmail A GmailService StructuredBipEmail (must contain
    *   BIP_FIELD_KEYS.GMAIL_MESSAGE_ID, GMAIL_THREAD_ID, SOURCE_EMAIL_SUBJECT and
    *   SOURCE_EMAIL_RECEIVED_AT).
    * @param {Object} extractedFields The normalized object returned by
    *   GeminiService.extractBipFieldsFromEmail(), keyed by
    *   BIP_CONSTANTS.GEMINI_EXTRACTED_FIELD_KEYS.
    * @param {string[]=} knownFacultyNames Known STU faculty names, used to resolve
    *   BIP_FIELD_KEYS.SUITABLE_FACULTIES.
    * @return {BipRecordDto}
    * @throws {BIP_UTILS.AppError} VALIDATION_ERROR if required fields are missing
    *   or malformed.
    */
    function parseEmail(structuredEmail, extractedFields, knownFacultyNames) {
      if (!BIP_UTILS.isPlainObject(structuredEmail)) {
        throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'structuredEmail must be a plain object.');
      }
      if (!BIP_UTILS.isPlainObject(extractedFields)) {
        throw BIP_UTILS.createError(ERROR_CODES.VALIDATION_ERROR, 'extractedFields must be a plain object produced by GeminiService.');
      }
      assertCompleteExtraction_(extractedFields);

      const gmailMessageId = BIP_UTILS.requireNonEmptyString(
        structuredEmail[FIELD_KEYS.GMAIL_MESSAGE_ID],
        FIELD_KEYS.GMAIL_MESSAGE_ID
      );
      const gmailThreadId = BIP_UTILS.requireNonEmptyString(
        structuredEmail[FIELD_KEYS.GMAIL_THREAD_ID],
        FIELD_KEYS.GMAIL_THREAD_ID
      );

      const bipTitle = BIP_UTILS.requireNonEmptyString(
        normalizeBipTitle_(extractedFields[FIELD_KEYS.BIP_TITLE]),
        FIELD_KEYS.BIP_TITLE
      );
      const university = BIP_UTILS.requireNonEmptyString(
        normalizeUniversityName(extractedFields[FIELD_KEYS.UNIVERSITY]),
        FIELD_KEYS.UNIVERSITY
      );
      const country = BIP_UTILS.requireNonEmptyString(
        normalizeCountry(extractedFields[FIELD_KEYS.COUNTRY]),
        FIELD_KEYS.COUNTRY
      );
      const applicationMethod = BIP_UTILS.requireNonEmptyString(
        normalizeApplicationMethod_(extractedFields[FIELD_KEYS.APPLICATION_METHOD]),
        FIELD_KEYS.APPLICATION_METHOD
      );

      const academicLevel = normalizeAcademicLevel(extractedFields[FIELD_KEYS.ACADEMIC_LEVEL]);
      const additionalNotes = normalizeAdditionalNotes_(extractedFields[FIELD_KEYS.ADDITIONAL_NOTES]);

      const startDate = normalizeIsoDate_(extractedFields[FIELD_KEYS.START_DATE], FIELD_KEYS.START_DATE, false);
      const endDate = normalizeIsoDate_(extractedFields[FIELD_KEYS.END_DATE], FIELD_KEYS.END_DATE, false);
      const applicationDeadline = normalizeIsoDate_(
        extractedFields[FIELD_KEYS.APPLICATION_DEADLINE],
        FIELD_KEYS.APPLICATION_DEADLINE,
        true
      );
      assertChronologicalOrder_(startDate, endDate);

      const resolvedFaculties = resolveSuitableFaculties(extractedFields[FIELD_KEYS.SUITABLE_FACULTIES], knownFacultyNames);

      const sourceEmailSubject = BIP_UTILS.isNonEmptyString(structuredEmail[FIELD_KEYS.SOURCE_EMAIL_SUBJECT])
        ? structuredEmail[FIELD_KEYS.SOURCE_EMAIL_SUBJECT].trim()
        : '';
      const sourceEmailReceivedAt = (structuredEmail[FIELD_KEYS.SOURCE_EMAIL_RECEIVED_AT] instanceof Date)
        ? structuredEmail[FIELD_KEYS.SOURCE_EMAIL_RECEIVED_AT]
        : new Date();

      const record = {};
      record[FIELD_KEYS.BIP_TITLE] = bipTitle;
      record[FIELD_KEYS.UNIVERSITY] = university;
      record[FIELD_KEYS.COUNTRY] = country;
      record[FIELD_KEYS.ACADEMIC_LEVEL] = academicLevel;
      record[FIELD_KEYS.START_DATE] = startDate;
      record[FIELD_KEYS.END_DATE] = endDate;
      record[FIELD_KEYS.APPLICATION_DEADLINE] = applicationDeadline;
      record[FIELD_KEYS.APPLICATION_METHOD] = applicationMethod;
      record[FIELD_KEYS.SUITABLE_FACULTIES] = resolvedFaculties.join(SUITABLE_FACULTIES_JOIN_SEPARATOR_);
      record[FIELD_KEYS.ADDITIONAL_NOTES] = additionalNotes;

      record[FIELD_KEYS.GMAIL_MESSAGE_ID] = gmailMessageId;
      record[FIELD_KEYS.GMAIL_THREAD_ID] = gmailThreadId;
      record[FIELD_KEYS.SOURCE_EMAIL_SUBJECT] = sourceEmailSubject;
      record[FIELD_KEYS.SOURCE_EMAIL_RECEIVED_AT] = sourceEmailReceivedAt;
      record[FIELD_KEYS.IMPORTED_AT] = new Date();

      BIP_UTILS.logInfo('ParserService', 'Built validated BIP record DTO from Gemini extraction.', {
        gmailMessageId,
        bipTitle: record[FIELD_KEYS.BIP_TITLE]
      });

      return record;
    }

    return {
      parseEmail,
      normalizeAcademicLevel,
      normalizeCountry,
      normalizeUniversityName,
      resolveSuitableFaculties
    };

  })());

NotificationService.gs

  /**
  * @fileoverview Reminder-email notification layer for the BIP Automation
  * system.
  *
  * NotificationService.gs merges BIP record data and Faculty Coordinator
  * data into the "Email Templates" sheet's reminder template and sends
  * the resulting email. It is the only file responsible for composing
  * and dispatching reminder emails and for recording that a reminder was
  * sent.
  *
  * ARCHITECTURAL POSITION (Layer 3 - External Data Sources / Communication)
  * Depends on Constants.gs, Utilities.gs (BIP_UTILS), Config.gs
  * (ConfigService) for the reminder interval and sender display name, and
  * SpreadsheetService.gs (read/write) for templates, coordinators and
  * reminder bookkeeping. It never touches SpreadsheetApp, DriveApp or
  * GmailApp directly - all persistence flows through SpreadsheetService.
  *
  * MAIL DELIVERY NOTE
  * GmailService.gs is the sole file permitted to interact with GmailApp,
  * and it exposes no send capability (by design, it is a read-only
  * ingestion service). Outbound reminder emails are therefore sent via
  * Apps Script's MailApp service, which is a distinct API from GmailApp
  * and is appropriate for simple transactional notification email.
  *
  * REMINDER TEMPLATE NAME NOTE
  * The literal "Template" column value identifying the reminder template
  * in the "Email Templates" sheet is not yet part of the confirmed sheet
  * mapping in Constants.gs. REMINDER_TEMPLATE_NAME_ below is therefore
  * the single place that literal value is defined; if the real template
  * name differs, update only that constant.
  *
  * GLOBAL SCOPE NOTE
  * Exposes exactly ONE global symbol, `NotificationService`, built via an
  * IIFE, to avoid collisions in the shared Apps Script global namespace.
  */

  const NotificationService = Object.freeze((() => {

    const FIELD_KEYS = BIP_CONSTANTS.BIP_FIELD_KEYS;
    const COORD_KEYS = BIP_CONSTANTS.COORDINATOR_FIELD_KEYS;
    const PLACEHOLDER_TOKENS = BIP_CONSTANTS.PLACEHOLDER_TOKENS;
    const ERROR_CODES = BIP_CONSTANTS.ERROR_CODES;

    /**
    * The "Template" column value in the "Email Templates" sheet that
    * identifies the reminder email template. See the REMINDER TEMPLATE
    * NAME NOTE above.
    * @private
    */
    const REMINDER_TEMPLATE_NAME_ = 'BIP Reminder';

    /**
    * Splits a stored, comma-separated "Suitable Faculties" cell value back
    * into an array of trimmed faculty names.
    * @param {*} facultiesValue
    * @return {string[]}
    * @private
    */
    function splitFacultyList_(facultiesValue) {
      if (typeof facultiesValue !== 'string' || facultiesValue.trim() === '') {
        return [];
      }
      return facultiesValue.split(',').map((name) => name.trim()).filter((name) => name !== '');
    }

    /**
    * Replaces every BIP_CONSTANTS.PLACEHOLDER_TOKENS occurrence in `text`
    * with the corresponding value from `mergeData`.
    * @param {string} text Template subject or HTML body.
    * @param {Object<string, string>} mergeData Map of field key -> merge value.
    * @param {boolean} escapeHtmlValues Whether to HTML-escape merged values (true for HTML bodies).
    * @return {string}
    * @private
    */
    function mergeText_(text, mergeData, escapeHtmlValues) {
      let merged = typeof text === 'string' ? text : '';
      Object.keys(mergeData).forEach((fieldKey) => {
        const token = PLACEHOLDER_TOKENS[fieldKey];
        if (!token) {
          return;
        }
        const rawValue = mergeData[fieldKey];
        const value = escapeHtmlValues ? BIP_UTILS.escapeHtml(rawValue) : rawValue;
        merged = merged.split(token).join(value);
      });
      return merged;
    }

    /**
    * Builds the flat merge-data map (field key -> string value) for a
    * given BIP record and the coordinator being notified.
    * @param {Object} record A BIP record as returned by SpreadsheetService.
    * @param {Object} coordinator A coordinator record as returned by SpreadsheetService.
    * @return {Object<string, string>}
    * @private
    */
    function buildMergeData_(record, coordinator) {
      const mergeData = {};

      Object.keys(FIELD_KEYS).forEach((constName) => {
        const fieldKey = FIELD_KEYS[constName];
        const value = record[fieldKey];
        mergeData[fieldKey] = (value === undefined || value === null) ? '' : String(value);
      });

      Object.keys(COORD_KEYS).forEach((constName) => {
        const fieldKey = COORD_KEYS[constName];
        const value = coordinator[fieldKey];
        mergeData[fieldKey] = (value === undefined || value === null) ? '' : String(value);
      });

      return mergeData;
    }

    /**
    * Resolves the deduplicated set of faculty coordinators (by email) that
    * should be notified for a given BIP record, based on its stored
    * SUITABLE_FACULTIES value.
    * @param {Object} record
    * @return {Object[]} Coordinator records, each keyed by BIP_CONSTANTS.COORDINATOR_FIELD_KEYS.
    * @private
    */
    function collectUniqueCoordinatorsForRecord_(record) {
      const facultyNames = splitFacultyList_(record[FIELD_KEYS.SUITABLE_FACULTIES]);
      const uniqueByEmail = {};

      facultyNames.forEach((facultyName) => {
        let matches;
        try {
          matches = SpreadsheetService.getFacultyCoordinatorsByFaculty(facultyName);
        } catch (err) {
          BIP_UTILS.logWarn('NotificationService', 'Failed to look up coordinators for a suitable faculty; skipping it.', {
            facultyName,
            error: String(err)
          });
          return;
        }
        matches.forEach((coordinator) => {
          const emailKey = String(coordinator[COORD_KEYS.COORDINATOR_EMAIL]).trim().toLowerCase();
          if (emailKey !== '' && !uniqueByEmail[emailKey]) {
            uniqueByEmail[emailKey] = coordinator;
          }
        });
      });

      return Object.keys(uniqueByEmail).map((emailKey) => uniqueByEmail[emailKey]);
    }

    /**
    * Merges and sends a single reminder email to one coordinator.
    * @param {Object} coordinator
    * @param {Object} record
    * @param {{subject: string, body: string}} template
    * @throws {BIP_UTILS.AppError} EXTERNAL_API_ERROR if MailApp fails to send.
    * @private
    */
    function dispatchReminderEmail_(coordinator, record, template) {
      const mergeData = buildMergeData_(record, coordinator);
      const subject = mergeText_(template.subject, mergeData, false);
      const htmlBody = mergeText_(template.body, mergeData, true);
      const coordinatorEmail = coordinator[COORD_KEYS.COORDINATOR_EMAIL];

      try {
        MailApp.sendEmail({
          to: coordinatorEmail,
          subject,
          htmlBody,
          name: ConfigService.getNotificationSenderName()
        });
      } catch (err) {
        throw BIP_UTILS.createError(
          ERROR_CODES.EXTERNAL_API_ERROR,
          `Unable to send reminder email to "${coordinatorEmail}" for Gmail message "${record[FIELD_KEYS.GMAIL_MESSAGE_ID]}": ${err.message || err}`
        );
      }

      BIP_UTILS.logInfo('NotificationService', 'Sent BIP reminder email.', {
        gmailMessageId: record[FIELD_KEYS.GMAIL_MESSAGE_ID],
        coordinatorEmail
      });
    }

    /**
    * Sends reminder emails for a single BIP record to every resolved
    * coordinator, then records that a reminder was sent for that record
    * exactly once (regardless of how many coordinators were notified).
    * @param {Object} record
    * @param {{subject: string, body: string}} template
    * @return {{gmailMessageId: string, recipientsNotified: number, recipients: string[]}}
    * @throws {BIP_UTILS.AppError} Propagates any dispatch or persistence failure.
    * @private
    */
    function processRecordReminder_(record, template) {
      const gmailMessageId = record[FIELD_KEYS.GMAIL_MESSAGE_ID];
      const coordinators = collectUniqueCoordinatorsForRecord_(record);

      if (coordinators.length === 0) {
        BIP_UTILS.logWarn('NotificationService', 'No faculty coordinators could be resolved for BIP record; skipping reminder.', {
          gmailMessageId,
          suitableFaculties: record[FIELD_KEYS.SUITABLE_FACULTIES]
        });
        return { gmailMessageId, recipientsNotified: 0, recipients: [] };
      }

      const recipients = [];
      coordinators.forEach((coordinator) => {
        dispatchReminderEmail_(coordinator, record, template);
        recipients.push(coordinator[COORD_KEYS.COORDINATOR_EMAIL]);
      });

      const sentAtIso = BIP_UTILS.getCurrentIsoTimestamp();
      SpreadsheetService.recordReminderSent(gmailMessageId, sentAtIso);

      return { gmailMessageId, recipientsNotified: recipients.length, recipients };
    }

    return {

      /**
      * Sends reminder emails for every BIP record currently eligible for
      * one, per SpreadsheetService.getRecordsPendingReminder() and the
      * configured reminder interval (ConfigService.getReminderIntervalDays()).
      * A failure processing one record does not stop the batch; it is
      * captured in the returned `failures` array.
      * @return {{totalCandidates: number, recordsNotified: number, emailsSent: number, failures: Array<{gmailMessageId: string, error: string, errorCode: string}>}}
      */
      sendPendingReminders() {
        const intervalDays = ConfigService.getReminderIntervalDays();
        const pendingRecords = SpreadsheetService.getRecordsPendingReminder(intervalDays);

        const result = {
          totalCandidates: pendingRecords.length,
          recordsNotified: 0,
          emailsSent: 0,
          failures: []
        };

        if (pendingRecords.length === 0) {
          BIP_UTILS.logInfo('NotificationService', 'No BIP records are currently due for a reminder.', {});
          return result;
        }

        const template = SpreadsheetService.getEmailTemplate(REMINDER_TEMPLATE_NAME_);

        pendingRecords.forEach((record) => {
          try {
            const outcome = processRecordReminder_(record, template);
            if (outcome.recipientsNotified > 0) {
              result.recordsNotified += 1;
              result.emailsSent += outcome.recipientsNotified;
            }
          } catch (err) {
            BIP_UTILS.logError('NotificationService', `Failed to send reminder for Gmail message "${record[FIELD_KEYS.GMAIL_MESSAGE_ID]}".`, {
              error: err.message
            });
            result.failures.push({
              gmailMessageId: record[FIELD_KEYS.GMAIL_MESSAGE_ID],
              error: err.message,
              errorCode: (err instanceof BIP_UTILS.AppError) ? err.errorCode : ERROR_CODES.INTERNAL_ERROR
            });
          }
        });

        BIP_UTILS.logInfo('NotificationService', 'Completed reminder dispatch run.', result);
        return result;
      },

      /**
      * Sends a reminder immediately for a single, explicitly identified
      * BIP record, bypassing the interval eligibility check. Intended for
      * the ROUTER_ACTIONS.SEND_REMINDER manual trigger.
      * @param {string} gmailMessageId
      * @return {{gmailMessageId: string, recipientsNotified: number, recipients: string[]}}
      * @throws {BIP_UTILS.AppError} NOT_FOUND if no matching record exists.
      */
      sendReminderForRecord(gmailMessageId) {
        BIP_UTILS.requireNonEmptyString(gmailMessageId, 'gmailMessageId');

        const record = SpreadsheetService.findRecordByGmailMessageId(gmailMessageId);
        if (!record) {
          throw BIP_UTILS.createError(
            ERROR_CODES.NOT_FOUND,
            `No BIP record found for Gmail message "${gmailMessageId}".`
          );
        }

        const template = SpreadsheetService.getEmailTemplate(REMINDER_TEMPLATE_NAME_);
        const outcome = processRecordReminder_(record, template);

        BIP_UTILS.logInfo('NotificationService', 'Manually triggered reminder dispatch completed.', outcome);
        return outcome;
      }
    };

  })());

AutomationService.gs

  /**
  * @fileoverview Top-level workflow orchestrator for the BIP Automation
  * system.
  *
  * AutomationService.gs coordinates the full pipeline:
  *   Gmail -> (duplicate check) -> Gemini -> Parser -> Spreadsheet -> Drive
  * for new BIP invitation emails, and separately:
  *   Spreadsheet -> Notification
  * for reminder dispatch. It contains no business logic of its own beyond
  * sequencing calls, interpreting their results, and isolating per-item
  * failures so a single bad email cannot abort an entire run - all actual
  * domain logic lives in the individual services it calls.
  *
  * ARCHITECTURAL POSITION (Layer 4 - Orchestration)
  * Depends on Constants.gs, Utilities.gs (BIP_UTILS), Config.gs
  * (ConfigService, for exclusive-execution locking), GmailService.gs,
  * GeminiService.gs, ParserService.gs, SpreadsheetService.gs,
  * DriveService.gs and NotificationService.gs. It never calls GmailApp,
  * SpreadsheetApp, DriveApp or UrlFetchApp directly - every external
  * interaction flows through the corresponding service.
  *
  * DUPLICATE-CHECK NOTE
  * The "duplicate check" pipeline stage is already fully implemented
  * inside GmailService.searchNewBipEmails() (Gmail label exclusion plus a
  * per-message check against SpreadsheetService), so AutomationService
  * does not - and must not - re-implement it; it simply consumes the
  * already-deduplicated result. A second, narrower duplicate case
  * (DUPLICATE_ENTRY thrown by SpreadsheetService.appendBipRecord, e.g. a
  * thread that was fetched but never labeled on a prior run) is handled
  * here defensively.
  *
  * CONCURRENCY NOTE
  * Both the import pipeline and the reminder pipeline run inside
  * ConfigService.runExclusive() so that overlapping scheduled-trigger or
  * manual-Router-action executions cannot interleave.
  *
  * TRIGGER HANDLER NOTE
  * runScheduledImport() and runScheduledReminders(), declared at the
  * bottom of this file, are plain global functions whose names exactly
  * match BIP_CONSTANTS.TRIGGER_HANDLERS.RUN_SCHEDULED_IMPORT and
  * RUN_SCHEDULED_REMINDERS. Apps Script installable time-based triggers
  * invoke handlers by global function name, so these thin wrappers - not
  * AutomationService's own methods - are what a TriggerService must
  * register triggers against.
  *
  * GLOBAL SCOPE NOTE
  * Exposes exactly ONE global object symbol, `AutomationService`, built
  * via an IIFE, plus the two global trigger-handler functions described
  * above, to avoid collisions in the shared Apps Script global namespace.
  */

  const AutomationService = Object.freeze((() => {

    const FIELD_KEYS = BIP_CONSTANTS.BIP_FIELD_KEYS;
    const COORD_KEYS = BIP_CONSTANTS.COORDINATOR_FIELD_KEYS;
    const ERROR_CODES = BIP_CONSTANTS.ERROR_CODES;

    /**
    * Builds the deduplicated list of known STU faculty names from the
    * "Faculty Coordinators" sheet, used by ParserService to resolve each
    * BIP record's suitable-faculties field.
    * @return {string[]}
    * @private
    */
    function getKnownFacultyNames_() {
      const coordinators = SpreadsheetService.getFacultyCoordinators();
      const uniqueNames = {};
      coordinators.forEach((coordinator) => {
        const facultyName = coordinator[COORD_KEYS.FACULTY_NAME];
        if (BIP_UTILS.isNonEmptyString(facultyName)) {
          uniqueNames[facultyName] = true;
        }
      });
      return Object.keys(uniqueNames);
    }

    /**
    * Marks a Gmail message as processed without letting a labeling
    * failure abort the pipeline; a future run's sheet-level duplicate
    * check still protects against re-importing it.
    * @param {Object} structuredEmail
    * @private
    */
    function markProcessedSafely_(structuredEmail) {
      try {
        GmailService.markEmailAsProcessed(structuredEmail[FIELD_KEYS.GMAIL_MESSAGE_ID]);
      } catch (err) {
        BIP_UTILS.logWarn('AutomationService', 'Failed to label Gmail message as processed; it may be re-processed on the next run.', {
          gmailMessageId: structuredEmail[FIELD_KEYS.GMAIL_MESSAGE_ID],
          error: String(err)
        });
      }
    }

    /**
    * Archives the source email's HTML content and any attachments into
    * the record's Drive folder. Individual archival failures are logged
    * and swallowed so they never block a successful import.
    * @param {string} folderId
    * @param {Object} structuredEmail
    * @private
    */
    function archiveEmailContentToDrive_(folderId, structuredEmail) {
      const fileName = `${BIP_UTILS.sanitizeForFileName(structuredEmail[FIELD_KEYS.SOURCE_EMAIL_SUBJECT])}.html`;

      try {
        DriveService.saveTextFileToFolder(
          folderId,
          fileName,
          structuredEmail.htmlBody || structuredEmail.plainBody || '',
          'text/html'
        );
      } catch (err) {
        BIP_UTILS.logWarn('AutomationService', 'Failed to archive source email HTML to Drive; continuing.', {
          gmailMessageId: structuredEmail[FIELD_KEYS.GMAIL_MESSAGE_ID],
          error: String(err)
        });
      }

      const attachments = Array.isArray(structuredEmail.attachments) ? structuredEmail.attachments : [];
      attachments.forEach((attachment) => {
        try {
          DriveService.saveBlobToFolder(folderId, attachment.blob);
        } catch (err) {
          BIP_UTILS.logWarn('AutomationService', 'Failed to archive an email attachment to Drive; continuing.', {
            gmailMessageId: structuredEmail[FIELD_KEYS.GMAIL_MESSAGE_ID],
            fileName: attachment.fileName,
            error: String(err)
          });
        }
      });
    }

    /**
    * Runs the full Gemini -> Parser -> Spreadsheet -> Drive pipeline for a
    * single structured email. Never throws: all failures are captured in
    * the returned outcome so batch processing can continue.
    * @param {Object} structuredEmail A GmailService StructuredBipEmail.
    * @param {string[]} knownFacultyNames
    * @return {{gmailMessageId: string, status: string, bipTitle: ?string, driveFolderUrl: ?string, error: ?string}}
    *   status is one of 'imported', 'duplicate', 'failed'.
    * @private
    */
    function processSingleEmail_(structuredEmail, knownFacultyNames) {
      const gmailMessageId = structuredEmail[FIELD_KEYS.GMAIL_MESSAGE_ID];

      try {
        const extractedFields = GeminiService.extractBipFieldsFromEmail({
          subject: structuredEmail[FIELD_KEYS.SOURCE_EMAIL_SUBJECT],
          fromAddress: structuredEmail.fromAddress,
          body: structuredEmail.plainBody,
          plainBody: structuredEmail.plainBody
        });

        const record = ParserService.parseEmail(structuredEmail, extractedFields, knownFacultyNames);

        let persistedRecord;
        try {
          persistedRecord = SpreadsheetService.appendBipRecord(record);
        } catch (appendError) {
          if (appendError instanceof BIP_UTILS.AppError && appendError.errorCode === ERROR_CODES.DUPLICATE_ENTRY) {
            markProcessedSafely_(structuredEmail);
            BIP_UTILS.logWarn('AutomationService', 'Skipped duplicate BIP record during import.', { gmailMessageId });
            return {
              gmailMessageId,
              status: 'duplicate',
              bipTitle: record[FIELD_KEYS.BIP_TITLE],
              driveFolderUrl: null,
              error: null
            };
          }
          throw appendError;
        }

        const folderInfo = DriveService.createBipRecordFolder(
          persistedRecord[FIELD_KEYS.UNIVERSITY],
          persistedRecord[FIELD_KEYS.BIP_TITLE]
        );

        archiveEmailContentToDrive_(folderInfo.folderId, structuredEmail);
        SpreadsheetService.setDriveFolderUrl(gmailMessageId, folderInfo.folderUrl);
        markProcessedSafely_(structuredEmail);

        BIP_UTILS.logInfo('AutomationService', 'Successfully imported BIP record.', {
          gmailMessageId,
          bipTitle: persistedRecord[FIELD_KEYS.BIP_TITLE],
          folderUrl: folderInfo.folderUrl
        });

        return {
          gmailMessageId,
          status: 'imported',
          bipTitle: persistedRecord[FIELD_KEYS.BIP_TITLE],
          driveFolderUrl: folderInfo.folderUrl,
          error: null
        };
      } catch (err) {
        BIP_UTILS.logError('AutomationService', `Failed to process BIP email "${gmailMessageId}".`, { error: err.message });
        return {
          gmailMessageId,
          status: 'failed',
          bipTitle: null,
          driveFolderUrl: null,
          error: err.message
        };
      }
    }

    return {

      /**
      * Runs the complete import pipeline: searches Gmail for new,
      * not-yet-processed BIP invitation emails and, for each one, extracts
      * structured fields via Gemini, builds a validated record via
      * ParserService, persists it via SpreadsheetService, and creates its
      * Drive folder via DriveService. Runs under an exclusive script lock.
      * @param {number=} maxResults Maximum number of new emails to process (passed through to GmailService).
      * @return {{candidatesFound: number, imported: number, duplicates: number, failed: number, results: Array<Object>}}
      */
      runImportPipeline(maxResults) {
        return ConfigService.runExclusive(() => {
          const startedAt = Date.now();
          const summary = {
            candidatesFound: 0,
            imported: 0,
            duplicates: 0,
            failed: 0,
            results: []
          };

          const structuredEmails = GmailService.searchNewBipEmails(maxResults);
          summary.candidatesFound = structuredEmails.length;

          if (structuredEmails.length === 0) {
            BIP_UTILS.logInfo('AutomationService', 'Import pipeline found no new BIP emails.', {});
            return summary;
          }

          const knownFacultyNames = getKnownFacultyNames_();

          structuredEmails.forEach((structuredEmail) => {
            const outcome = processSingleEmail_(structuredEmail, knownFacultyNames);
            summary.results.push(outcome);
            if (outcome.status === 'imported') {
              summary.imported += 1;
            } else if (outcome.status === 'duplicate') {
              summary.duplicates += 1;
            } else {
              summary.failed += 1;
            }
          });

          BIP_UTILS.logInfo('AutomationService', 'Import pipeline run completed.', {
            candidatesFound: summary.candidatesFound,
            imported: summary.imported,
            duplicates: summary.duplicates,
            failed: summary.failed,
            durationMs: Date.now() - startedAt
          });

          return summary;
        });
      },

      /**
      * Runs the full pipeline for exactly one, explicitly identified Gmail
      * message, regardless of whether it was previously labeled as
      * processed. Intended for the ROUTER_ACTIONS.PARSE manual action.
      * Runs under an exclusive script lock.
      * @param {string} gmailMessageId
      * @return {{gmailMessageId: string, status: string, bipTitle: ?string, driveFolderUrl: ?string, error: ?string}}
      */
      runManualParse(gmailMessageId) {
        BIP_UTILS.requireNonEmptyString(gmailMessageId, 'gmailMessageId');
        return ConfigService.runExclusive(() => {
          const structuredEmail = GmailService.getEmailByMessageId(gmailMessageId.trim());
          const knownFacultyNames = getKnownFacultyNames_();
          return processSingleEmail_(structuredEmail, knownFacultyNames);
        });
      },

      /**
      * Runs the reminder pipeline via NotificationService: either a
      * single, explicitly identified record (when gmailMessageId is
      * provided) or every record currently eligible per the configured
      * reminder interval. Runs under an exclusive script lock. Intended
      * for the ROUTER_ACTIONS.SEND_REMINDER action.
      * @param {string=} gmailMessageId Optional single record to notify.
      * @return {Object} NotificationService.sendReminderForRecord() or sendPendingReminders() result.
      */
      runReminderPipeline(gmailMessageId) {
        return ConfigService.runExclusive(() => {
          if (BIP_UTILS.isNonEmptyString(gmailMessageId)) {
            return NotificationService.sendReminderForRecord(gmailMessageId.trim());
          }
          return NotificationService.sendPendingReminders();
        });
      }
    };

  })());

  /**
  * Global time-based trigger handler for the scheduled import job. Name
  * must exactly match BIP_CONSTANTS.TRIGGER_HANDLERS.RUN_SCHEDULED_IMPORT.
  * @return {Object} AutomationService.runImportPipeline() result.
  */
  function runScheduledImport() {
    return AutomationService.runImportPipeline();
  }

  /**
  * Global time-based trigger handler for the scheduled reminder job. Name
  * must exactly match BIP_CONSTANTS.TRIGGER_HANDLERS.RUN_SCHEDULED_REMINDERS.
  * @return {Object} AutomationService.runReminderPipeline() result.
  */
  function runScheduledReminders() {
    return AutomationService.runReminderPipeline();
  }
